using System.Collections;
using System.Text.Json;
using SemiconductorCsvAnalyzer.Models;
using SemiconductorCsvAnalyzer.Services;

static class DiskStorageProbe
{
    static int checks;
    static void Check(bool value, string message)
    { if (!value) throw new Exception(message); checks++; Console.WriteLine("PASS " + message); }
    private sealed class DescendingValues(int count) : IReadOnlyList<TestValue>
    {
        public int Count => count;
        public TestValue this[int index] => new() { Value = count - index };
        public IEnumerator<TestValue> GetEnumerator() { for (int i = 0; i < Count; i++) yield return this[i]; }
        IEnumerator IEnumerable.GetEnumerator() => GetEnumerator();
    }
    static CsvParseResult Fixture(int chips, bool disk)
    {
        var records = new RecordList(0, disk);
        for (int pass = 0; pass < 4; pass++)
            for (int chip = 0; chip < chips; chip++)
            {
                var row = new CsvRecord { Id = $"C{chip}R{pass}", SerialNumber = $"{pass * chips + chip + 1}",
                    WaferId = "W" + chip % 2, X = chip, Y = 0, Site = "S" + chip % 4, HBin = pass == 3 ? "1" : "2", SBin = "1" };
                for (int item = 0; item < 5; item++)
                {
                    if ((chip % 50 + pass + item) % 7 == 0) continue;
                    row.Measurements.Add("T" + item, (chip % 50 + pass + item) % 13 == 0 ? "invalid\"文本" :
                        (chip % 50 + pass + item) % 11 == 0 ? "" : ((chip % 50) + pass * .125 + item).ToString("R", System.Globalization.CultureInfo.InvariantCulture));
                }
                records.Add(row);
            }
        return DatasetService.Build(Enumerable.Range(0, 5).Select(i => new TestItemData { Id = "T" + i, Name = "T" + i, TestNumber = (i + 1).ToString(), LowLimit = 0, HighLimit = 30 }).ToList(), records);
    }
    public static int Run(string folder)
    {
        checks = 0;
        using var operation = new AnalysisOperation();
        using (var vector = PagedVector<long>.From(Enumerable.Range(0, 40000).Select(i => (long)i * 13)))
        {
            Check(vector.IsOnDisk && vector[8191] == 8191L * 13 && vector[8192] == 8192L * 13 && vector[^1] == 39999L * 13, "vector exact values across page boundaries");
            vector[8192] = -42; DiskStorage.Trim(); Check(vector[8192] == -42, "vector writes survive cache eviction");
            using var shared = PagedVector<long>.Open(vector.Capture()); Check(shared.SequenceEqual(vector), "readonly vector snapshot shares exact file contents");
        }
        using (var keys = DiskKeySet.From(new[] { "one", "two", "文本" }))
        using (var other = DiskKeySet.From(new[] { "two", "three" }))
        { keys.UnionWith(other); Check(keys.Count == 4 && keys.Contains("three") && keys.Contains("文本"), "disk membership SQL union preserves ordinal keys and count"); }
        var vectors = Enumerable.Range(0, 200).Select(_ => PagedVector<long>.From(Enumerable.Range(0, 8000).Select(i => (long)i))).ToArray();
        Check(DiskStorage.SmallVectorBytes <= DiskStorage.SmallVectorBudget && vectors.Any(v => v.IsOnDisk) && vectors.All(v => v[7999] == 7999), "many small column indices share a process-wide spill budget");
        foreach (var vector in vectors) vector.Dispose();
        var small = Fixture(50, false); var large = Fixture(2100, true);
        using var lease = new DatasetLease(large.Records);
        Check(large.Records.IsOnDisk && large.Items.All(i => i.Values.IsDiskBacked), "identity and all numeric columns spill from first imported record");
        foreach (RetestMode mode in Enum.GetValues<RetestMode>())
        {
            var expected = DatasetService.SelectRetests(small, mode); var actual = DatasetService.SelectRetests(large, mode);
            using var viewLease = new DatasetLease(actual.Records);
            var reference = expected.Records.ToDictionary(r => r.Id);
            bool same = actual.Records.All(row =>
            {
                string id = $"C{row.X!.Value % 50}R{int.Parse(row.Id.Split('R')[1])}";
                return reference.TryGetValue(id, out var truth) && row.Measurements.OrderBy(v => v.Key).SequenceEqual(truth.Measurements.OrderBy(v => v.Key));
            });
            Check(actual.Records.Count == expected.Records.Count * 42 && same, "disk partial retest matches original raw/presence selection " + mode);
            Check(actual.Items.Zip(expected.Items).All(p => p.First.Values.Count == p.Second.Values.Count * 42 && p.First.Errors.Count == p.Second.Errors.Count * 42), "disk sparse numeric/error counts " + mode);
            using var manifest = DiskDatasetSnapshot.Capture(actual);
            var restored = JsonSerializer.Deserialize<DiskDatasetSnapshot>(JsonSerializer.Serialize(manifest))!.Open();
            Check(restored.Items.Zip(actual.Items).All(p => p.First.Values.Select(v => v.Value).SequenceEqual(p.Second.Values.Select(v => v.Value))) &&
                restored.Items.Zip(actual.Items).All(p => p.First.Errors.Select(r => r.Id).SequenceEqual(p.Second.Errors.Select(r => r.Id))), "readonly worker manifest preserves projection, numeric values and invalid rows " + mode);
        }
        DiskStorage.Trim();
        Check(large.Records[64].Id == "C64R0" && large.Records[4096].Id == "C1996R1", "record identity boundaries remain exact after eviction");
        large.Records[65].SerialNumber = "reassigned"; large.Records.Flush(); DiskStorage.Trim();
        Check(large.Records[65].SerialNumber == "reassigned" && large.Records.Store!.RunLookup(serial: "reassigned") == 65, "record updates persist metadata and SN index together");
        Check(large.Records.Store!.RunLookup(wafer: "W0", x: 64, y: 0) == 6364 && large.Records.Store.RunLookup(wafer: "W0", after: 64) == 66, "disk Run finds latest XY and acquisition-order neighbors");
        foreach (ScatterOrder order in Enum.GetValues<ScatterOrder>())
        {
            var values = large.Items[0].Values;
            var sorted = ScatterSeries.BuildSorted(values, 0, 30, order, waferOrder: new[] { "W0", "W1" }, trackChips: true);
            var reference = Enumerable.Range(0, values.Count).OrderBy(i => order == ScatterOrder.Wafer ? int.Parse(values[i].WaferId[1..]) : 0)
                .ThenBy(i => order == ScatterOrder.Site ? NumericText.SortKey(values[i].Site) : 0)
                .ThenBy(i => order == ScatterOrder.Site ? values[i].Site : "", StringComparer.OrdinalIgnoreCase)
                .ThenBy(i => order == ScatterOrder.Site ? values[i].Site : "", StringComparer.Ordinal)
                .ThenBy(i => NumericText.SortKey(values[i].SerialNumber)).ThenBy(i => values[i].SerialNumber, StringComparer.OrdinalIgnoreCase).ThenBy(i => i).ToArray();
            Check(sorted.Points.IsOnDisk && sorted.Points.Select(p => p.SourceIndex).SequenceEqual(reference), "external scatter sort is stable with all points " + order);
            var marks = new ScatterChipMarks(); marks.AddRectangle(sorted, 1, 50, double.MinValue, double.MaxValue);
            using var mask = marks.CreatePagedMask(sorted); Check(mask != null && mask.Count == sorted.Points.Count && mask.Any(b => b == 1), "large scatter marks use paged membership " + order);
        }
        var summary1 = SummaryService.Build(large.Items, new[] { "S0", "S1", "S2", "S3" }, true, new());
        DiskStorage.Trim();
        var summary2 = SummaryService.Build(large.Items, new[] { "S0", "S1", "S2", "S3" }, true, new());
        Check(summary1.Zip(summary2).All(p => p.First.Count == p.Second.Count && p.First.Mean == p.Second.Mean && p.First.Median == p.Second.Median), "disk summary result cache survives page eviction");
        int resources = large.Records.Resources.Count;
        SummaryRows published; IDisposable summaryLease;
        using (var preparation = new AnalysisOperation())
        {
            published = new SummaryRows(2500);
            for (int i = 0; i < published.Count; i++) published[i] = new() { Data = large.Items[0], Site = i.ToString(), Mean = i };
            summaryLease = published.AcquireLease();
        }
        Check(published[2499].Mean == 2499 && large.Records.Resources.Count == resources, "published summary lease survives preparation without accumulating stores on source dataset");
        var statisticsStore = typeof(SummaryRows).GetField("_store", System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic)!.GetValue(published)!;
        string statisticsPath = (string)statisticsStore.GetType().GetField("_path", System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic)!.GetValue(statisticsStore)!;
        summaryLease.Dispose(); Check(!File.Exists(statisticsPath), "replaced summary table releases its disk store independently of source data");
        var altered = SummaryService.Build(large.Items, new[] { "S0" }, true, new() { RemoveOutliers = false });
        Check(altered.All(r => r.RobustMean == r.Mean), "statistics cache includes outlier configuration");
        var ordered = ExternalValueSort.Sort(Enumerable.Range(0, ExternalValueSort.RunValues + 177).Select(i => (double)(-i)));
        double prior = double.NegativeInfinity; int total = 0; bool inOrder = true;
        foreach (double value in ordered) { inOrder &= value >= prior; prior = value; total++; }
        Check(inOrder && total == ExternalValueSort.RunValues + 177, "exact external median sort crosses run boundary without lost values");
        int medianCount = ExternalValueSort.RunValues + 2;
        Check(StatisticsService.CalculateMedian(new DescendingValues(medianCount)) == (medianCount + 1) * .5, "large-column median remains exact across spilled runs");
        var diskSimulation = SimulationSnapshot.Build(large, new(), RetestMode.Last, default);
        var smallSimulation = SimulationSnapshot.Build(small, new(), RetestMode.Last, default);
        Check(diskSimulation.Count == 2100 && diskSimulation.States.Select((states, item) => states.Select((state, chip) => state == smallSimulation.States[item][chip % 50]).All(v => v)).All(v => v), "simulation state matrix streams chip histories to disk");
        Check(BinStatisticsService.Build(large.Records, RetestMode.Last, new()).ChipCount == 2100 &&
            WaferStatisticsService.Build(large.Records, RetestMode.Last, new()).ChipCounts.Sum() == 2100, "whole-chip BIN and wafer counts use disk grouping");
        using (var rows = new PagedObjectRows<ManagedFileResult>(Enumerable.Range(0, 6000).Select(i => new ManagedFileResult { FileId = "F" + i, Records = i })))
        { DiskStorage.Trim(); Check(rows.Count == 6000 && rows[4096].FileId == "F4096" && rows[5999].Records == 5999, "management table metadata pages remain exact"); }
        Management(folder);
        string cancelledPath = Path.Combine(folder, "disk-cancel.csv");
        using (var writer = new StreamWriter(cancelledPath))
        {
            writer.WriteLine("SN,X,Y,Site,SBIN,HBIN,Wafer,T\nLSL,,,,,,,0\nUSL,,,,,,,2\nUnit,,,,,,,V");
            for (int i = 0; i < 100000; i++) writer.WriteLine($"{i + 1},{i},0,1,1,1,W,1");
        }
        string cache = Path.GetDirectoryName(DiskStorage.NewPath(".probe"))!;
        var priorFiles = Directory.EnumerateFiles(cache, Environment.ProcessId + "-*").ToHashSet();
        bool cancelled = false;
        using (var cancellation = new CancellationTokenSource())
        using (var cancelledOperation = new AnalysisOperation(cancellation.Token))
        {
            cancellation.CancelAfter(30);
            try { CsvParser.Parse(cancelledPath, new() { TestItemRow = 1, LowLimitRow = 2, HighLimitRow = 3, UnitRow = 4,
                DataStartRow = 5, TestItemStartColumn = 8, SerialNumberColumn = 1, XColumn = 2, YColumn = 3, SiteColumn = 4,
                SBinColumn = 5, HBinColumn = 6, WaferColumn = 7 }, null, cancellation.Token); }
            catch (OperationCanceledException) { cancelled = true; }
        }
        Check(cancelled && !Directory.EnumerateFiles(cache, Environment.ProcessId + "-*").Except(priorFiles).Any(), "mid-import cancellation releases all newly constructed structured cache files");
        DiskStorage.ConfigureBudget(256); Check(DiskStorage.CachedBytes <= DiskStorage.CacheBudgetBytes, "global data page cache respects shared byte budget");
        Console.WriteLine($"DISK STORAGE: {checks} CHECKS PASSED"); return checks;
    }
    static void Management(string folder)
    {
        const int chips = 5000;
        var config = new AnalyzerConfig { TestItemRow = 1, LowLimitRow = 2, HighLimitRow = 3, UnitRow = 4, DataStartRow = 5,
            TestItemStartColumn = 8, SerialNumberColumn = 1, XColumn = 2, YColumn = 3, SiteColumn = 4, SBinColumn = 5, HBinColumn = 6, WaferColumn = 7 };
        var product = new ManagedProduct { Name = "Disk management", ConfigText = ConfigLoader.Serialize(config), Flow = new() {
            new() { Id = "cold", Name = "cold", Kind = "CP", Temperature = -40 }, new() { Id = "hot", Name = "hot", Kind = "CP", Temperature = 125 } } };
        for (int pass = 0; pass < 3; pass++)
        {
            string path = Path.Combine(folder, "disk-management-" + pass + ".csv");
            using (var writer = new StreamWriter(path))
            {
                writer.WriteLine("SN,X,Y,Site,SBIN,HBIN,Wafer,T\nLSL,,,,,,,0\nUSL,,,,,,,2\nUnit,,,,,,,V");
                for (int chip = 0; chip < chips; chip++) writer.WriteLine($"{chip + 1},{chip},0,1,1,{(pass == 0 && chip % 2 == 0 ? 2 : 1)},W,1");
            }
            product.Files.Add(new() { Id = "F" + pass, StageId = pass == 1 ? "hot" : "cold", Lot = "L", FilePath = path, TestTime = new DateTime(2026, pass + 1, 1) });
        }
        product.RetestMode = RetestMode.First; var first = ProductYieldService.Calculate(product);
        var combined = first.Yields.Single(r => r.IsCombined);
        Check(combined.Total == chips && combined.Good == chips / 2 && combined.Failed == chips / 2, "management disk cross-temperature first retest matches full-chip truth");
        product.RetestMode = RetestMode.Last; var last = ProductYieldService.Calculate(product);
        combined = last.Yields.Single(r => r.IsCombined);
        Check(combined.Total == chips && combined.Good == chips && last.Files.All(f => f.Records == chips && f.Error.Length == 0), "management disk last retest handles files beyond in-memory threshold");
        Check(last.MonthlyYields.Single(r => r.IsCombined).Period == "2026-03" && combined.SourceFileIds.SetEquals(new[] { "F0", "F1", "F2" }), "management disk matching preserves monthly assignment and source dependencies");
        string bad = Path.Combine(folder, "disk-management-invalid.csv");
        File.WriteAllText(bad, "SN,X,Y,Site,SBIN,HBIN,Wafer,T\nLSL,,,,,,,0\nUSL,,,,,,,2\nUnit,,,,,,,V\n1,9999,0,1,1,1,W,1\n2,10000,0,1,1,1,,1\n");
        product.Files.Add(new() { Id = "bad", StageId = "cold", Lot = "L", FilePath = bad, WaferOverride = "override" });
        var failed = ProductYieldService.Calculate(product);
        Check(failed.Files.Single(f => f.FileId == "bad").Error.Length > 0 && failed.Yields.Single(r => r.IsCombined).Total == chips, "failed management file rolls back its entire staged population");
    }
}
