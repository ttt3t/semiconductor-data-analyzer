using System.Diagnostics;
using System.IO;
using System.Reflection;
using System.Runtime.InteropServices;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Threading;
using SemiconductorCsvAnalyzer;
using SemiconductorCsvAnalyzer.Models;
using SemiconductorCsvAnalyzer.Services;

static class Probe
{
    [DllImport("user32.dll", CharSet = CharSet.Unicode)] static extern nint FindWindow(string? className, string title);
    [DllImport("user32.dll")] static extern uint GetWindowThreadProcessId(nint window, out uint process);
    [DllImport("user32.dll")] static extern bool PostMessage(nint window, uint message, nint value, nint extra);
    static void Pump() { var frame = new DispatcherFrame(); Dispatcher.CurrentDispatcher.BeginInvoke(DispatcherPriority.ApplicationIdle, new Action(() => frame.Continue = false)); Dispatcher.PushFrame(frame); }
    [STAThread] static int Main(string[] args)
    {
        try
        {
            string folder = Path.GetFullPath(args[0]);
            var config = new AnalyzerConfig { TestItemRow = 1, LowLimitRow = 2, HighLimitRow = 3, UnitRow = 4, TestNumberRow = 5,
                SBinRow = 0, HBinRow = 0, DataStartRow = 6, TestItemStartColumn = 8, SerialNumberColumn = 1, XColumn = 2,
                YColumn = 3, SiteColumn = 4, SBinColumn = 5, HBinColumn = 6, WaferColumn = 7 };
            var sources = Directory.GetFiles(folder, "wafer-*.csv").OrderBy(p => p, StringComparer.Ordinal).Select((p, i) => new CsvImportSource(p, i == 0 ? null : new(false, "", false, false))).ToArray();
            if (sources.Length != 25) throw new Exception("Expected 25 batch fixtures.");
            var app = new App(); app.InitializeComponent(); app.ShutdownMode = ShutdownMode.OnExplicitShutdown;
            SynchronizationContext.SetSynchronizationContext(new DispatcherSynchronizationContext());
            var failures = new List<Exception>(); app.DispatcherUnhandledException += (_, e) => { failures.Add(e.Exception); e.Handled = true; Console.Error.WriteLine(e.Exception); };
            var window = new MainWindow(new LayoutSettings(Path.Combine(folder, "ui-layout.json")), new RecentFilesService(Path.Combine(folder, "ui-recent.json")), new UiPreferencesService(Path.Combine(folder, "ui-prefs.json")));
            typeof(MainWindow).GetField("_config", BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(window, config);
            typeof(MainWindow).GetField("_managedConfigLoaded", BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(window, true);
            window.Show(); Pump();
            // Each CSV starts its SN at 1. A real batch correctly displays the
            // existing renumbering notice; acknowledge only that expected dialog
            // in this probe's process so the integration test can continue.
            int notices = 0;
            int ticks = 0;
            using var acknowledge = new System.Threading.Timer(_ =>
            {
                if (Interlocked.Increment(ref ticks) % 40 == 0)
                {
                    var current = (CsvParseResult)typeof(MainWindow).GetField("_data", BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(window)!;
                    Console.WriteLine($"UI batch pending: {current.Records.Count} committed records, notices={Volatile.Read(ref notices)}");
                }
                nint dialog = FindWindow(null, "Run 序列号已自动生成");
                if (dialog == 0) return;
                GetWindowThreadProcessId(dialog, out uint process);
                if (process == Environment.ProcessId) { Interlocked.Increment(ref notices); Console.WriteLine("ACK expected serial renumbering notice"); PostMessage(dialog, 0x0111, 1, 0); }
            }, null, 50, 50);
            var clock = Stopwatch.StartNew();
            var task = (Task)typeof(MainWindow).GetMethod("OpenBatchAsync", BindingFlags.Instance | BindingFlags.NonPublic)!.Invoke(window, new object[] { sources })!;
            while (!task.IsCompleted) { if (clock.Elapsed.TotalSeconds > 180) throw new TimeoutException("Batch UI did not finish."); Pump(); Thread.Sleep(2); }
            task.GetAwaiter().GetResult(); Pump(); acknowledge.Change(Timeout.Infinite, Timeout.Infinite);
            var data = (CsvParseResult)typeof(MainWindow).GetField("_data", BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(window)!;
            var grid = (DataGrid)window.FindName("DgTestItems");
            if (data.Records.Count != 50000 || data.Items.Count != 120 || grid.Items.Count != 12000 || failures.Count != 0 || notices == 0) throw new Exception("Incomplete batch UI results or missing serial-number notice.");
            Console.WriteLine($"PASS actual OpenBatchAsync: 25 CSV / 50000 rows / 120 tests / 12000 grouped summaries, {clock.Elapsed.TotalSeconds:F3}s");
            typeof(MainWindow).GetField("_retestMode", BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(window, RetestMode.Last);
            clock.Restart();
            task = (Task)typeof(MainWindow).GetMethod("RefreshTestItemList", BindingFlags.Instance | BindingFlags.NonPublic)!.Invoke(window, null)!;
            while (!task.IsCompleted) { if (clock.Elapsed.TotalSeconds > 180) throw new TimeoutException("Retest refresh did not finish."); Pump(); Thread.Sleep(2); }
            task.GetAwaiter().GetResult(); Pump();
            if (!ReferenceEquals(typeof(MainWindow).GetField("_allItems", BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(window), data.Items) || grid.Items.Count != 12000 || failures.Count != 0) throw new Exception("Unique-chip Last mode rebuilt its columns.");
            Console.WriteLine($"PASS unique-chip Last mode reuses source columns and restores 12000 summary rows, {clock.Elapsed.TotalSeconds:F3}s");
            window.Close(); Pump(); if (failures.Count != 0) throw new Exception("Batch close raised dispatcher errors.");
            app.Shutdown(); return 0;
        }
        catch (Exception exception) { Console.Error.WriteLine(exception); return 1; }
    }
}
