    static int checks;
    static readonly List<string> checkFailures = new();
    static void Check(bool ok, string message) { checks++; if (!ok) checkFailures.Add(message); Console.WriteLine((ok ? "PASS " : "FAIL ") + message); }
    static object? Field(object target, string name) => target.GetType().GetField(name, BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(target);
    static void Set(object target, string name, object value) => target.GetType().GetField(name, BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(target, value);
    static object? Call(object target, string name, params object[] args) => target.GetType().GetMethod(name, BindingFlags.Instance | BindingFlags.NonPublic)!.Invoke(target, args);
    static void Pump() { var frame = new DispatcherFrame(); Dispatcher.CurrentDispatcher.BeginInvoke(DispatcherPriority.ApplicationIdle, new Action(() => frame.Continue = false)); Dispatcher.PushFrame(frame); }
    static void Wait(Task task) { var watch = Stopwatch.StartNew(); while (!task.IsCompleted) { if (watch.Elapsed.TotalSeconds > 60) throw new TimeoutException(); Pump(); Thread.Sleep(2); } task.GetAwaiter().GetResult(); Pump(); }
    static BitmapSource Png(Stream stream) { stream.Position = 0; return BitmapDecoder.Create(stream, BitmapCreateOptions.PreservePixelFormat, BitmapCacheOption.OnLoad).Frames[0]; }
    static byte[] Pixels(BitmapSource bitmap) { var b = new FormatConvertedBitmap(bitmap, PixelFormats.Bgra32, null, 0); byte[] result = new byte[b.PixelWidth * b.PixelHeight * 4]; b.CopyPixels(result, b.PixelWidth * 4, 0); return result; }
    static bool Same(BitmapSource a, BitmapSource b) => a.PixelWidth == b.PixelWidth && a.PixelHeight == b.PixelHeight && Pixels(a).SequenceEqual(Pixels(b));
    static bool SameReport(BitmapSource a, BitmapSource b)
    {
        if (a.PixelWidth != b.PixelWidth || a.PixelHeight != b.PixelHeight) return false;
        var expected = Pixels(a); var actual = Pixels(b); int changed = 0;
        for (int i = 0; i < expected.Length; i += 4)
        {
            if (expected.AsSpan(i, 4).SequenceEqual(actual.AsSpan(i, 4))) continue;
            // WPF can shade one edge pixel differently for a fractional-size heading on another STA.
            // Keep plot data/axes exact; allow only four small grayscale differences in the heading row.
            int y = i / 4 / a.PixelWidth;
            if (y >= Math.Ceiling(26 * a.DpiY / 96) || ++changed > 4 ||
                expected[i] != expected[i + 1] || expected[i] != expected[i + 2] ||
                actual[i] != actual[i + 1] || actual[i] != actual[i + 2]) return false;
            for (int channel = 0; channel < 4; channel++) if (Math.Abs(expected[i + channel] - actual[i + channel]) > 24) return false;
        }
        return true;
    }
    static void Copy(MainWindow main, string method) => Call(main, method, main, new RoutedEventArgs());
    static string Status(MainWindow main) => ((TextBlock)main.FindName("TxtStatus")).Text;

    static HistogramView? View(object target, string property) =>
        (HistogramView?)target.GetType().GetProperty(property, BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(target);

    static int CountOutside(IEnumerable<HistogramBin> bins) => bins.Where(bin => bin.IsOutOfSpec).Sum(bin => bin.Count);

    static bool CutsAt(IEnumerable<HistogramBin> bins, double limit) =>
        bins.All(bin => !(bin.From < limit && bin.To > limit));

    static void WaitForHistogram(MainWindow main)
    {
        var timer = Stopwatch.StartNew();
        while ((bool)Field(main, "_detailPending")!)
        {
            Pump();
            Thread.Sleep(2);
            if (timer.Elapsed > TimeSpan.FromSeconds(15)) throw new TimeoutException("Histogram detail request");
        }
        Pump();
    }

    static void CheckHistogram(MainWindow main, HistogramControl histogram, string folder)
    {
        var legacy = ConfigLoader.Parse("[Histogram]\nBinCount=47");
        Check(Enumerable.Range(0, 3).Select(legacy.HistogramBinCount).SequenceEqual(new[] { 47, 150, 80 }),
            "legacy INI preserves fixed bins and both original mode defaults");
        var settings = new SettingsWindow(legacy, _ => { });
        ((TextBox)settings.FindName("TxtBinCount")).Text = "40";
        ((TextBox)settings.FindName("TxtSpecZoneBinCount")).Text = "17";
        ((TextBox)settings.FindName("TxtMedianBinCount")).Text = "13";

        settings.Close();

            "each mode can be edited independently in settings and survives INI round-trip");

        var samples = Enumerable.Range(-50, 101).Select(value => new TestValue { Value = value }).ToArray();
        var spec = HistogramService.CreateSpecZoneBins(samples, -20, 20, false, config.SpecZoneBinCount);
        Check(spec.Count == 17 && spec.Sum(bin => bin.Count) == 41 && CountOutside(spec) == 0 &&
            spec.Any(bin => bin.From == -20) && spec.Any(bin => bin.To == 20),
            "spec-zone bins keep both threshold boundaries and include observations equal to either limit");
        var flat = HistogramService.CreateFixedBins(new[] { new TestValue { Value = 9 } }, 40, null, null, true);
        Check(flat.Count == 40 && flat.Sum(bin => bin.Count) == 1 && flat[0].From < 9 && flat[^1].To > 9,
            "constant-valued data still renders the configured number of bins");
        var mixedSamples = new[] { 8.5, 9.0, 9.5, 10.0, 10.5 }.Select(value => new TestValue { Value = value }).ToArray();
        var mixed = HistogramService.CreateFixedBins(mixedSamples, 1, 9, 10, true);
        var excludingOutliers = HistogramService.CreateFixedBins(mixedSamples, 1, 9, 10, false);
        double strictLow = Math.BitIncrement(9), strictHigh = Math.BitDecrement(10);
        var strict = HistogramService.CreateFixedBins(mixedSamples, 1, strictLow, strictHigh, true);
        Check(mixed.Select(bin => bin.Count).SequenceEqual(new[] { 1, 3, 1 }) &&
            mixed.Select(bin => bin.IsOutOfSpec).SequenceEqual(new[] { true, false, true }) &&
            CutsAt(mixed, 9) && CutsAt(mixed, 10) &&
            excludingOutliers is [{ Count: 3, IsOutOfSpec: false }] &&
            strict.Select(bin => bin.Count).SequenceEqual(new[] { 2, 1, 2 }) &&
            CutsAt(strict, strictLow) && CutsAt(strict, strictHigh),
            "each threshold cuts the viewport even with only one requested bin, including strict STDF endpoints");
        var oneSided = HistogramService.CreateFixedBins(mixedSamples, 1, null, 9.5, true);
        var coincident = HistogramService.CreateFixedBins(mixedSamples, 1, 9.5, 9.5, true);
        Check(oneSided.Count == 2 && CountOutside(oneSided) == 2 && CutsAt(oneSided, 9.5) &&
            coincident.Single(bin => !bin.IsOutOfSpec) is { From: 9.5, To: 9.5, Count: 1 } &&
            CountOutside(coincident) == 4,
            "a one-sided limit makes a single cut and identical inclusive limits preserve the exact point");

        config.LowLimitRow = 2; config.HighLimitRow = 3; config.UnitRow = 0; config.DataStartRow = 4;
        config.TestItemStartColumn = 8; config.SerialNumberColumn = 1;
        config.SBinColumn = 5; config.HBinColumn = 6; config.WaferColumn = 7;
        string csv = Path.Combine(folder, "histogram-range.csv");
        File.WriteAllLines(csv, new[] { "SN,X,Y,Site,SBIN,HBIN,Wafer,Measurement", "LSL,,,,,,,-20", "USL,,,,,,,20" }
            .Concat(Enumerable.Range(-50, 101).Select((value, index) =>
                $"SN{index},{index},0,1,1,1,W1,{(value == 0 ? "0.25" : value.ToString(CultureInfo.InvariantCulture))}")));
        Wait((Task)Call(main, "ApplySettings", config)!);
        Wait((Task)Call(main, "OpenSelectedFilesAsync", (object)new[] { csv })!);
        ((DataGrid)main.FindName("DgTestItems")).SelectedIndex = 0;
        Call(main, "SetPlotView", "Histogram");
        WaitForHistogram(main);
        bool HasRange(int count, double from, double to, int samples) => histogram.Bins is { } bins &&
            bins.Count == count && Math.Abs(bins[0].From - from) < 1e-9 && Math.Abs(bins[^1].To - to) < 1e-9 &&

        var scatter = Field(main, "_displayedScatter");
        Call(main, "ChangeHistogramRange", 2d);
        WaitForHistogram(main);
        Check(HasRange(40, -100, 100, 101), "zoom out rebins the larger range into the same 40 bins");
        Call(main, "ChangeHistogramRange", .4d);
        WaitForHistogram(main);
        Check(HasRange(40, -20, 20, 41) && View(histogram, "View") is { Below: 30, Above: 30 } &&

            ((TextBlock)main.FindName("TxtUnit")).Text.Contains("Count=101"),

        Call(main, "ChangeHistogramRange", 1d);
        WaitForHistogram(main);
        Check(HasRange(40, -50, 50, 101), "reset restores the original range and frequencies");

        Point PlotCenter(Rect plot) => new(plot.Left + plot.Width / 2, plot.Top + plot.Height / 2);
        var plot = PlotBounds();

        Check((bool)Call(main, "BeginHistogramDrag", dragStart)! && histogram.IsMouseCaptured,
            "starting a left drag within the chart captures the pointer");

        Call(main, "MoveHistogramDrag", dragStart.X - plot.Width / 4);
        WaitForHistogram(main);
        Check(HasRange(40, -25, 75, 76), "moving left updates both x-axis endpoints and rebins the latest range");
        Call(main, "EndHistogramDrag");

        var wheel = new MouseWheelEventArgs(Mouse.PrimaryDevice, Environment.TickCount, 120)
            { RoutedEvent = UIElement.PreviewMouseWheelEvent };
        histogram.RaiseEvent(wheel);
        WaitForHistogram(main);
        double zoomRadius = 50d / 1.2;
        Check(wheel.Handled && HasRange(40, 25 - zoomRadius, 25 + zoomRadius, 67),
            "plain mouse wheel zooms around the center created by dragging");
        Call(main, "ChangeHistogramRange", .4d);
        WaitForHistogram(main);
        Check(HasRange(40, 5, 45, 41) && CutsAt(histogram.Bins!, 20) && CountOutside(histogram.Bins!) == 25 &&
            ReferenceEquals(scatter, Field(main, "_displayedScatter")),
            "zoom preserves the panned center and the visible one-sided cut with exact frequencies");

        plot = PlotBounds();

        Call(main, "MoveHistogramDrag", dragStart.X + plot.Width / 2);
        Call(main, "EndHistogramDrag");
        WaitForHistogram(main);
        Check(HasRange(40, -15, 25, 41) && !histogram.IsMouseCaptured,
            "moving right when zoomed uses the displayed width and releases capture");
        Call(main, "UpdateHistogramRange", 1d, 0d);
        WaitForHistogram(main);
        Check(HasRange(40, -50, 50, 101), "reset clears both the scale and the panned center");

        var modes = (ListBox)main.FindName("CmbBinMode");
        modes.SelectedIndex = 1;
        WaitForHistogram(main);
        Check(histogram.Bins?.Count == 17, "switching to spec zone uses its configured bin count");
        modes.SelectedIndex = 2;

        Check(histogram.Bins?.Count == 13, "switching to median center uses its configured bin count");
        var counts = (CheckBox)main.FindName("ChkHistogramCounts");
        main.UpdateLayout(); Pump();
        var size = ChartImage.SizeOf(histogram);
        var unlabelled = ChartImage.Capture(histogram, size);
        counts.IsChecked = true; Pump();
        Check(histogram.ShowBinCounts && !Same(unlabelled, ChartImage.Capture(histogram, size)),
            "the header checkbox turns on visible frequency labels in the live histogram");

        var autoLog = (CheckBox)main.FindName("ChkHistogramAutoLogFrequency");
        Check(autoLog.IsChecked == true && histogram.AutoLogFrequency, "automatic log frequency is enabled by default");
        counts.IsChecked = false;
        BitmapSource DrawFrequencies(params int[] frequencies)
        {
            histogram.Apply(frequencies.Select((count, index) => new HistogramBin
                { From = index, To = index + 1, Count = count }).ToList(),
                null, null, null, null, false, false, false, false);
            Pump();
            return ChartImage.Capture(histogram, size);
        }
        var atBoundary = DrawFrequencies(0, 1, 100);
        var singleNonzero = DrawFrequencies(0, 0, 1000);
        var log = DrawFrequencies(0, 1, 101);
        autoLog.IsChecked = false;
        var linear = DrawFrequencies(0, 1, 101);
        plot = PlotBounds();
        bool ContainsSmallBar(BitmapSource image)
        {
            int x = (int)((plot.Left + plot.Width / 2) * image.DpiX / 96);
            int y = (int)((plot.Bottom - plot.Height * .08) * image.DpiY / 96);
            var pixels = Pixels(image);
            int offset = (y * image.PixelWidth + x) * 4;
            return pixels[offset] == 180 && pixels[offset + 1] == 130 && pixels[offset + 2] == 70;
        }
        Check(ContainsSmallBar(log) && !ContainsSmallBar(linear),
            "a ratio over 100:1 raises the small bar on a log scale and unchecking restores linear height");
        Check(Same(atBoundary, DrawFrequencies(0, 1, 100)) && Same(singleNonzero, DrawFrequencies(0, 0, 1000)),
            "exactly 100:1 and lone positive bins with zeros keep a linear scale");
        autoLog.IsChecked = true;
        DrawFrequencies(0, 1, 101);
        Check(Same(atBoundary, DrawFrequencies(0, 1, 100)),
            "changing the displayed bins below the threshold returns to linear while automatic scaling stays enabled");

        histogram.Apply(mixed, null, null, null, null, false, false, false, false);
        Pump();
        var barPixels = Pixels(ChartImage.Capture(histogram, size));
        plot = PlotBounds();
        bool HasBarColor(double xRatio, double y, byte red, byte green, byte blue)
        {
            int x = (int)((plot.Left + plot.Width * xRatio) * size.DpiX / 96);
            int row = (int)((plot.Top + y * plot.Height) * size.DpiY / 96);
            int pixel = (row * size.PixelWidth + x) * 4;
            return barPixels[pixel] == blue && barPixels[pixel + 1] == green && barPixels[pixel + 2] == red;
        }
        Call(histogram, "UpdateFrequencyToolTip", PlotCenter(plot));
        string hint = ((ToolTip)histogram.ToolTip).Content.ToString()!;
        Check(HasBarColor(.5, .25, 70, 130, 180) && HasBarColor(.5, .75, 70, 130, 180) &&
            HasBarColor(.125, .8, 220, 90, 90) && hint.Contains("[9, 10]") &&
            hint.Contains("数量 3") && hint.Contains("规格内"),
            "threshold-separated bins use a single color and report the exact interval and count");

        var data = (CsvParseResult)Field(main, "_data")!;
        string report = Path.Combine(folder, "histogram-mode.xlsx");
        Wait(XlsxReportService.SaveAsync(report, data, config,
            new(data.Wafers, RetestMode.All, null, 1, true, new[] { data.Items[0].Id },
                HistogramSize: new(590, 230), ScatterSize: new(590, 230), HistogramShowBinCounts: true), "histogram-test"));
        using var book = new XLWorkbook(report);
        var sheet = book.Worksheet("关注测项");
        Check(sheet.Pictures.Count() == 2 && sheet.CellsUsed().Any(cell => cell.GetString().Contains("规格区间（17 箱）")),
            "XLSX report generates its histogram using the selected mode and bin count");

        string integerCsv = Path.Combine(folder, "integer-histogram.csv");
        File.WriteAllLines(integerCsv, new[] { "SN,X,Y,Site,SBIN,HBIN,Wafer,Measurement", "LSL,,,,,,,1.5", "USL,,,,,,,3" }
            .Concat(Enumerable.Range(1, 4).Select(value => $"SN{value},{value},0,1,1,1,W1,{value}")));
        Wait((Task)Call(main, "OpenSelectedFilesAsync", (object)new[] { integerCsv })!);
        ((DataGrid)main.FindName("DgTestItems")).SelectedIndex = 0;
        for (int mode = 0; mode < 3; mode++)
        {
            modes.SelectedIndex = mode;
            WaitForHistogram(main);
            var viewport = View(histogram, "View")!.Value;
            var integerBins = histogram.Bins!;
            int available = (int)(Math.Floor(viewport.To) - Math.Ceiling(viewport.From) + 1);
            var positions = Enumerable.Range(1, 4).ToArray();
            var visiblePositions = positions.Where(value => value >= viewport.From && value <= viewport.To).ToArray();
            Check(integerBins.Count == Math.Min(available, config.HistogramBinCount(mode)) &&
                integerBins.All(bin => double.IsInteger(bin.From) && double.IsInteger(bin.To) && bin.IncludesFrom && bin.IncludesTo) &&
                integerBins.Zip(integerBins.Skip(1)).All(pair => pair.First.To < pair.Second.From) &&
                integerBins.Sum(bin => bin.Count) == visiblePositions.Length &&
                CountOutside(integerBins) == visiblePositions.Count(value => value < 1.5 || value > 3) &&
                viewport.Below == positions.Count(value => value < viewport.From) &&
                viewport.Above == positions.Count(value => value > viewport.To) &&
                (mode != 1 || viewport.Above == 1) &&
                CutsAt(integerBins, 1.5) && CutsAt(integerBins, 3),
                "integer bins keep nonoverlapping closed intervals and exact frequencies in mode " + mode);
        }
        modes.SelectedIndex = 0;
        WaitForHistogram(main);
        Call(main, "UpdateHistogramRange", .001d, 0d);
        WaitForHistogram(main);
        Check(histogram.Bins is { Count: 0 } && View(histogram, "View") is { Below: 2, Above: 2 } && !PlotBounds().IsEmpty,
            "the integer viewport remains interactive when it contains no integers");
        Call(main, "UpdateHistogramRange", .001d, -.5d / 3);
        WaitForHistogram(main);
        Check(histogram.Bins!.Single() is { From: 2, To: 2, Count: 1, IsOutOfSpec: false } &&
            View(histogram, "View") is { Below: 1, Above: 2 },
            "panning into a single integer position restores its exact one-point frequency");

        var scaledItem = new TestItemData { DisplayFactor = 1000, LowLimit = .0015, HighLimit = .0035 };
        foreach (double value in new[] { .001, .002, .003, .004 })
            scaledItem.Values.Add(new TestValue { Value = value });
        var scaled = DetailCalculator.Calculate(new(scaledItem, null, 0, 2, true, true, true, true, true), default);
        Check(scaled.Bins.Count == 3 && scaled.Bins.All(bin => double.IsInteger(bin.From) && double.IsInteger(bin.To)) &&
            scaled.Bins.Sum(bin => bin.Count) == 4 && CountOutside(scaled.Bins) == 2 &&
            CutsAt(scaled.Bins, 1.5) && CutsAt(scaled.Bins, 3.5),
            "display-unit conversion preserves integer bins, allocating at least one bin to each threshold region");
        scaledItem.LowLimit = scaledItem.HighLimit = .003;
        var point = DetailCalculator.Calculate(new(scaledItem, null, 0, 1, true, true, true, true, true), default);
        Check(point.Bins.Single(bin => !bin.IsOutOfSpec) is { From: 3, To: 3, Count: 1 } && CountOutside(point.Bins) == 3,
            "a scaled inclusive endpoint stays in a singleton in-spec integer bin despite conversion rounding");

        string textLimitsCsv = Path.Combine(folder, "visible-text-limits.csv");
        File.WriteAllLines(textLimitsCsv, new[] { "SN,X,Y,Site,SBIN,HBIN,Wafer,Measurement",
            "LSL,,,,,,,Inf", "USL,,,,,,,3", "A1,0,0,1,1,1,W1,1", "A2,1,0,1,2,2,W1,4" });
        Wait((Task)Call(main, "OpenSelectedFilesAsync", (object)new[] { textLimitsCsv })!);
        var visible = ((DataGrid)main.FindName("DgTestItems")).Items.OfType<TestItemSummary>().Single();
        Check(visible.LslText == "Inf" && visible.Data.LowLimit == null && visible.Fails == 1,
            "the analyzer displays the source limit text and applies only the finite side");
        var textLimits = (CsvParseResult)Field(main, "_data")!;
        string textReport = Path.Combine(folder, "visible-text-limits.xlsx");
        Wait(XlsxReportService.SaveAsync(textReport, textLimits, config,
            new(textLimits.Wafers, RetestMode.All, null, 0, true, Array.Empty<string>()), "text-limit-test"));
        using var textBook = new XLWorkbook(textReport);
        var summarySheet = textBook.Worksheet("Testitem Yield");
        Check(summarySheet.Cell(8, 10).GetString() == "Inf" && summarySheet.Cell(8, 11).GetDouble() == 3 &&
            summarySheet.Cell(8, 13).GetDouble() == 1, "XLSX report shows the disabled text bound next to a numeric one-sided bound");
    }

    [STAThread] static int Main(string[] args)
    {
        try
        {
            if (args.Length == 3 && args[0] == "--hold-clipboard")
            {
                using var childReady = EventWaitHandle.OpenExisting(args[1]); using var childRelease = EventWaitHandle.OpenExisting(args[2]);
                bool opened = false;
                for (int i = 0; i < 100 && !(opened = OpenClipboard(0)); i++) Thread.Sleep(20);
                if (!opened) throw new System.ComponentModel.Win32Exception(Marshal.GetLastWin32Error());
                try { childReady.Set(); if (!childRelease.WaitOne(TimeSpan.FromSeconds(60))) throw new TimeoutException(); }
                finally { CloseClipboard(); }
                return 0;
            }
            string folder = Path.GetFullPath(args[0]); Directory.CreateDirectory(folder);
            var app = new App(); app.InitializeComponent(); app.ShutdownMode = ShutdownMode.OnExplicitShutdown;
            SynchronizationContext.SetSynchronizationContext(new DispatcherSynchronizationContext());
            var failures = new List<Exception>(); app.DispatcherUnhandledException += (_, e) => { failures.Add(e.Exception); e.Handled = true; };
            typeof(AppSettings).GetField("_userDirectory", BindingFlags.Static | BindingFlags.NonPublic)!.SetValue(null, folder);
            var main = new MainWindow(new LayoutSettings(Path.Combine(folder, "layout.json")), new RecentFilesService(Path.Combine(folder, "recent.json")), new UiPreferencesService(Path.Combine(folder, "ui.json")));
            main.Show(); Pump();
            var histogram = (HistogramControl)main.FindName("Histogram"); var scatter = (ScatterPlotControl)main.FindName("ScatterPlot");
            if (args.Length == 2 && args[1] == "--histogram")
            {
                CheckHistogram(main, histogram, folder);
                main.Close(); Pump(); Check(failures.Count == 0, "histogram operations cause no dispatcher exceptions");
                app.Shutdown();
                if (checkFailures.Count > 0) throw new Exception(string.Join(Environment.NewLine, checkFailures));
                Console.WriteLine($"ALL {checks} HISTOGRAM CHECKS PASSED"); return 0;
            }
            Check(((Button)main.FindName("BtnCopyHistogram")).IsVisible && ((Button)main.FindName("BtnCopyScatter")).IsVisible, "both chart headers expose copy buttons");
            var items = new List<TestItemData> { new() { Id = "a", Name = "A", DisplayName = "A", Unit = "V", LowLimit = 1, HighLimit = 3 }, new() { Id = "b", Name = "B", DisplayName = "B", Unit = "V", LowLimit = 1, HighLimit = 3 } };
            var records = Enumerable.Range(0, 6).Select(i => new CsvRecord { Id = "r" + i, SerialNumber = (6 - i).ToString(), WaferId = i % 2 == 0 ? "W2" : "W1", Site = i % 3 == 0 ? "2" : "1", X = i, Y = 0, HBin = "1", SBin = "1", Measurements = new() { ["a"] = (i + 1).ToString(), ["b"] = (i + 2).ToString() } }).ToList();

            scatter.SetSeries(ScatterSeries.BuildSorted(items[0].Values, 1, 3, ScatterOrder.SerialNumber, trackChips: true), 1, 3, "");
            histogram.Title = scatter.Title = "125_D_SNR_B_Min_1"; Pump();

            Check(scatter.ChipMarkCount == 1, "fixture marks one acquisition before copying");
            Call(scatter, "ZoomAt", At(3, 4), 120); Pump();
            var pixelBuffer = Field(scatter, "_pixelBuffer"); double scale = (double)Field(scatter, "_scaleX")!;
