using SemiconductorCsvAnalyzer.Models;
using SemiconductorCsvAnalyzer.Services;

static class BatchImportProbe
{
    static int checks;
    static void Check(bool condition, string message)
    { if (!condition) throw new Exception(message); checks++; Console.WriteLine("PASS " + message); }
    static AnalyzerConfig Config() => new() { TestItemRow = 1, TestNumberRow = 2, UnitRow = 3, LowLimitRow = 4, HighLimitRow = 5,
        SBinRow = 0, HBinRow = 0, DataStartRow = 6, TestItemStartColumn = 8, SerialNumberColumn = 1,
        XColumn = 2, YColumn = 3, SiteColumn = 4, SBinColumn = 5, HBinColumn = 6, WaferColumn = 7 };
    public static int Run(string root)
    {
        checks = 0; string folder = Path.Combine(root, "batch"); Directory.CreateDirectory(folder);
        using var operation = new AnalysisOperation();
        var config = Config(); var sources = new List<CsvImportSource>(); var originals = new List<string>();
        for (int file = 0; file < 25; file++)
        {
            bool missing = file == 3, swapped = file == 4;
            string path = Path.Combine(folder, file + ".csv");
            string pair(string a, string b) => missing ? a : swapped ? b + "," + a : a + "," + b;
            using (var writer = new StreamWriter(path))
            {
                writer.WriteLine("SN,X,Y,Site,SBIN,HBIN,Wafer," + pair("A", "B"));
                writer.WriteLine("Test Num,,,,,,," + pair("1", "2")); writer.WriteLine("Unit,,,,,,," + pair("V", "A"));
                writer.WriteLine("LSL,,,,,,," + pair("0", "0")); writer.WriteLine("USL,,,,,,," + pair(file == 24 ? "9" : "10", "10"));
                for (int i = 0; i < 129; i++) writer.WriteLine($"{i + 1},{i},0,{i % 2 + 1},1,1,W," + pair(i == 0 ? "0007" : "1.25", i == 0 ? "" : i == 1 ? "bad" : "2.25"));
            }
            originals.Add(File.ReadAllText(path));
            sources.Add(new(path, file == 0 ? null : new(false, "", false, file == 24)));
        }
        var data = DatasetService.LoadSources(sources, config); using var lease = new DatasetLease(data.Records);
        var a = data.Items.Single(i => i.Name == "A"); var b = data.Items.Single(i => i.Name == "B");
        Check(data.Records.Count == 3225 && data.Items.Count == 2 && data.Items.All(i => i.Values.IsDiskBacked), "25-file batch publishes all records with shared disk columns");
        Check(data.Wafers.SequenceEqual(Enumerable.Range(1, 25).Select(i => i == 1 ? "W" : $"W ({i})")), "batch preserves acquisition order and wafer collision suffixes");
        Check(a.Values.Count == 3225 && b.Values.Count == 24 * 127 && b.Errors.Count == 48 && a.HighLimit == 9, "batch remaps reordered/missing tests, invalid cells and last-file limits");
        Check(data.Records[0].Measurements[a.Id] == "0007" && data.Records[0].Measurements[b.Id] == "" && data.Records[129 + 1].Measurements[b.Id] == "bad" && !data.Records[3 * 129].Measurements.ContainsKey(b.Id), "batch preserves exact raw text, blank cells and absent measurements");
        Check(data.Records.Select(r => r.SerialNumber).SequenceEqual(Enumerable.Range(1, 3225).Select(i => i.ToString())) && data.Records[129].OriginalSerialNumber == "1", "batch normalizes display serials without changing original identities");
        Check(ReferenceEquals(DatasetService.SelectRetests(data, RetestMode.First), data) && ReferenceEquals(DatasetService.SelectRetests(data, RetestMode.Last), data) && DatasetService.SelectRetests(data, RetestMode.OnlyRetests).Records.Count == 0, "unique chips reuse disk data for First/Last and produce no retest-only records");
        var summaries = SummaryService.Build(data.Items, new[] { "1", "2" }, data.Wafers, false, false, config);
        Check(summaries.Sum(s => s.Count) == a.Values.Count + b.Values.Count && summaries.Sum(s => s.Eorr) == 48, "default grouped batch statistics include each finite value and error exactly once");
        Check(sources.Select((s, i) => File.ReadAllText(s.Path) == originals[i]).All(v => v), "batch import never changes source CSV bytes");
        using (var snapshot = DiskDatasetSnapshot.Capture(data))
        {
            var reopened = snapshot.Open(); using var reopenedLease = new DatasetLease(reopened.Records);
            Check(reopened.Records[24 * 129].Measurements[a.Id] == "0007" && reopened.Items[1].Values.ReadNumbers().SequenceEqual(b.Values.ReadNumbers()), "batch worker snapshot retains every raw store and remapped numeric column");
        }
        var original = DataFileService.Parse(sources[0].Path, config); using var originalLease = new DatasetLease(original.Records);
        string firstId = original.Records[0].Id;
        var appended = DatasetService.AppendSources(original, sources.Skip(1).Take(3).ToArray(), config);
        using var appendedLease = new DatasetLease(appended.Records);
        Check(original.Records.Count == 129 && original.Records[0].Id == firstId && original.Records[0].SerialNumber == "1" && original.Records[0].OriginalSerialNumber == null && original.Items[0].HighLimit == 10, "private accumulator detaches once and leaves the active dataset unchanged");
        bool failed = false;
        try { DatasetService.AppendSources(original, new[] { sources[1], new CsvImportSource(Path.Combine(folder, "missing.csv"), sources[1].Options) }, config); }
        catch (InvalidDataException) { failed = true; }
        Check(failed && original.Records.Count == 129 && original.Records[0].OriginalSerialNumber == null, "later batch failure preserves the complete active dataset");
        var same = DatasetService.LoadSources(new[] { sources[0], sources[1] with { Options = new(true, "W", false, false) } }, config);
        using var sameLease = new DatasetLease(same.Records);
        Check(same.Wafers.SequenceEqual(new[] { "W" }) && same.Devices.Count == 129 && DatasetService.SelectRetests(same, RetestMode.Last).Items.All(i => i.Values.Count + i.Errors.Count == 129), "same-wafer batch retains physical chip and partial-retest semantics");
        config.TestMode = TestAnalysisMode.FT; config.UuidTestNumbers = new() { "1" };
        var ft = DatasetService.LoadSources(sources.Take(2).ToArray(), config); using var ftLease = new DatasetLease(ft.Records);
        Check(ft.Records.All(r => r.IsFtIdentity) && ft.Records[0].RetestUuid == "0007" && ReferenceEquals(RetestIdentityService.Prepare(ft, config), ft), "unpublished FT identity preparation preserves leading zeroes without rebuilding a view");
        var paged = new RecordList(0, disk: true);
        for (int i = 0; i < 33; i++) paged.Add(new() { Id = "tail" + i, SerialNumber = i.ToString() });
        paged[32].SerialNumber = "changed";
        for (int i = 33; i < 130; i++) paged.Add(new() { Id = "tail" + i, SerialNumber = i.ToString() });
        paged.Flush(); DiskStorage.Trim();
        Check(paged[32].SerialNumber == "changed" && paged[33].Id == "tail33" && paged[129].Id == "tail129", "appending after a dirty partial page persists old edits and exposes every new row");
        Console.WriteLine($"BATCH IMPORT: {checks} CHECKS PASSED"); return checks;
    }
}
