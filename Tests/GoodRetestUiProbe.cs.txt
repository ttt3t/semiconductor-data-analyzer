    static int checks;
    static void Check(bool ok, string text) { if (!ok) throw new Exception(text); checks++; Console.WriteLine("PASS " + text); }
    static object? Field(object target, string name) => target.GetType().GetField(name, BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(target);
    static object? Call(object target, string name, params object[] args) => target.GetType().GetMethod(name, BindingFlags.Instance | BindingFlags.NonPublic)!.Invoke(target, args);
    static void Pump() { var f = new DispatcherFrame(); Dispatcher.CurrentDispatcher.BeginInvoke(DispatcherPriority.ApplicationIdle, new Action(() => f.Continue = false)); Dispatcher.PushFrame(f); }
    static void Wait(Task task) { var w = Stopwatch.StartNew(); while (!task.IsCompleted) { Pump(); Thread.Sleep(3); if (w.Elapsed.TotalSeconds > 60) throw new TimeoutException(); } task.GetAwaiter().GetResult(); Pump(); }
    static IEnumerable<T> Children<T>(DependencyObject root) where T : DependencyObject
    {
        for (int i = 0; i < VisualTreeHelper.GetChildrenCount(root); i++)
        {
            var child = VisualTreeHelper.GetChild(root, i);
            if (child is T item) yield return item;
            foreach (var nested in Children<T>(child)) yield return nested;
        }
    }
    static void CheckSBinFilter(MainWindow main, DataGrid grid, CsvParseResult original)
    {
        SampleFilterWindow Open()
        {
            Call(main, "SampleFilter_Click", main, new RoutedEventArgs());
            var window = Application.Current.Windows.OfType<SampleFilterWindow>().Single();
            ((DockPanel)window.Content).Children.OfType<TabControl>().Single().SelectedIndex = 1;
            window.UpdateLayout(); Pump();
            return window;
        }
        void Click(SampleFilterWindow window, string content)
        {
            Children<Button>(window).Single(button => Equals(button.Content, content)).RaiseEvent(new RoutedEventArgs(ButtonBase.ClickEvent));
            var timer = Stopwatch.StartNew();
            while (!window.IsEnabled)
            {
                if (timer.Elapsed > TimeSpan.FromSeconds(30)) throw new TimeoutException("SBIN selection");
                Pump(); Thread.Sleep(3);
            }
            Pump();
        }
        var filter = Open();
        var choices = (ListBox)Field(filter, "_sBinList")!;
        var checks = Children<CheckBox>(choices).ToArray();
        Check(checks.Select(box => box.Content).SequenceEqual(new object[] { "1", "2" }) &&
            checks.All(box => box.IsChecked == false), "SBIN tab lists loaded bin numbers and starts fully unchecked");
        checks[0].SetCurrentValue(ToggleButton.IsCheckedProperty, true); Pump();
        Check(choices.SelectedItems.Cast<string>().SequenceEqual(new[] { "1" }), "checkbox changes the SBIN selection");
        checks[1].SetCurrentValue(ToggleButton.IsCheckedProperty, true); Pump();
        Check(choices.SelectedItems.Count == 2, "multiple SBIN checkboxes can be selected together");
        checks[1].SetCurrentValue(ToggleButton.IsCheckedProperty, false); Pump();
        Click(filter, "应用");
        Check(((CsvParseResult)Field(main, "_scopeData")!).Records.Single().SBin == "1" &&
            grid.Items.OfType<TestItemSummary>().Single().Mean == 1 &&
            ((TextBlock)main.FindName("TxtSampleFilterAction")).Text == "数据筛选（1）" &&
            ReferenceEquals(original, Field(main, "_data")), "applying only SBIN filters the real analyzer and leaves loaded history available");
        Call(main, "SelectRun", 1); Pump();
        Check(grid.Items.OfType<TestItemSummary>().Single().RunValue == null, "Run Value is blank for a record excluded by SBIN");
        var scoped = Field(main, "_scopeData");
        Wait((Task)Call(main, "RefreshTestItemList")!);
        Check(ReferenceEquals(scoped, Field(main, "_scopeData")), "unchanged SBIN selection reuses the cached scope");
        filter.Close(); Pump();
        filter = Open();
        choices = (ListBox)Field(filter, "_sBinList")!;
        Check(choices.SelectedItems.Cast<string>().SequenceEqual(new[] { "1" }), "reopening restores the applied SBIN checks");
        checks = Children<CheckBox>(choices).ToArray();
        checks[0].SetCurrentValue(ToggleButton.IsCheckedProperty, false);
        checks[1].SetCurrentValue(ToggleButton.IsCheckedProperty, true); Pump();
        Click(filter, "应用");
        Check(((CsvParseResult)Field(main, "_scopeData")!).Records.Count == 2 &&
            grid.Items.OfType<TestItemSummary>().Single().Mean == 25,
            "changing SBIN with the same selection count replaces the cached sample scope");
        Click(filter, "取消数据筛选");
        Check(choices.SelectedItems.Count == 0 && ReferenceEquals(original, Field(main, "_scopeData")) &&
            grid.Items.OfType<TestItemSummary>().Single().Mean == 25, "clearing data filters restores all records and unchecks the SBIN tab");
        filter.Close(); Pump();
    }
    static void CheckRangeFilter(MainWindow main, DataGrid grid, CsvParseResult original)
    {
        void Until(Func<bool> ready)
        {
            var timer = Stopwatch.StartNew();
            while (!ready())
            {
                if (timer.Elapsed > TimeSpan.FromSeconds(20)) throw new TimeoutException("Measurement range filter");
                Pump(); Thread.Sleep(3);
            }
            Pump();
        }
        Call(main, "SampleFilter_Click", main, new RoutedEventArgs());
        var popup = Application.Current.Windows.OfType<SampleFilterWindow>().Single();
        popup.UpdateLayout(); Pump();
        var rows = (List<SampleFilterWindow.RangeRow>)Field(popup, "_rows")!;
        void Focus(bool value)
        {
            original.Items[0].Focus.IsFocused = value;
            Call(main, "Focus_Click", main, new RoutedEventArgs());
            Pump();
        }
        void Click(string content)
        {
            Children<Button>(popup).Single(button => Equals(button.Content, content)).RaiseEvent(new RoutedEventArgs(ButtonBase.ClickEvent));
            Until(() => popup.IsEnabled);
        }
        Check(rows.Count == 0, "measurement table starts empty when no test item is focused");
        Focus(true);
        var row = rows.Single();
        Check(row.Item.Id == original.Items[0].Id && row.Minimum == "" && row.Maximum == "",
            "focusing a measurement in the main analyzer immediately adds a blank row to the open table");
        row.Calculation = SampleCalculation.Relative;
        row.Reference = "20"; row.Minimum = "5"; row.Maximum = "5";
        Until(() => row.Preview.Contains("19 ~ 21"));
        Click("应用");
        Check(popup.Rules.Single().Deviation?.Reference == 20 &&
            ((CsvParseResult)Field(main, "_scopeData")!).Records.Count == 2 &&
            grid.Items.OfType<TestItemSummary>().Single().Mean == 20,
            "the relative interval preview matches the membership applied to the main analyzer");
        Call(main, "SelectRun", 0);
        row.Baseline = SampleBaseline.Golden;
        Click("将当前 Run 设为 Golden");
        ((RadioButton)Field(popup, "_outside")!).IsChecked = true;
        Until(() => row.Preview.Contains("范围外 0.95 ~ 1.05"));
        Click("应用");
        Check(popup.Rules.Single() is { Outside: true, Deviation.Reference: 1 } &&
            grid.Items.OfType<TestItemSummary>().Single().Mean == 25,
            "the selected raw Run supplies the Golden value and outside membership reaches the analyzer");
        Focus(false);
        Click("应用");
        Check(rows.Count == 0 && popup.Rules.Count == 0 && ReferenceEquals(original, Field(main, "_scopeData")),
            "unfocusing removes the row and Apply clears its measurement condition");
        popup.Close(); Pump();
    }
    static void CheckLocationFilter(MainWindow main, string folder, AnalyzerConfig config)
    {
        var path = Path.Combine(folder, "location-filter.csv");
        File.WriteAllText(path, "SN,X,Y,Site,SBIN,HBIN,Wafer,A,Lot\nLSL,,,,,,,0,\nUSL,,,,,,,100,\nUnit,,,,,,,V,\nTest Num,,,,,,,101,\n" +
            "1,0,0,1,1,1,W1,10,A\n2,1,0,1,3,3,W2,20,A\n3,,,1,2,2,W1,30,B\n4,0,0,1,4,4,W3,40,B\n5,0,0,1,9,9,W4,50,B\n");
        config.LotColumn = 9;
        Wait(main.OpenManagedFilesAsync(new[] { new CsvImportSource(path) }, config, RetestMode.All));
        var original = (CsvParseResult)Field(main, "_data")!;
        original.Items.Single().Focus.IsFocused = true;
        SampleFilterWindow Open()
        {
            Call(main, "SampleFilter_Click", main, new RoutedEventArgs());
            return Application.Current.Windows.OfType<SampleFilterWindow>().Single();
        }
        var popup = Open();
        void OpenTab(int index)
        {
            ((DockPanel)popup.Content).Children.OfType<TabControl>().Single().SelectedIndex = index;
            popup.UpdateLayout(); Pump();
        }
        ListBox List(string field) => (ListBox)Field(popup, field)!;
        CheckBox Box(string field, string label) => Children<CheckBox>(List(field)).Single(box => Equals(box.Content, label));
        void Click(string content)
        {
            Children<Button>(popup).Single(button => Equals(button.Content, content)).RaiseEvent(new RoutedEventArgs(ButtonBase.ClickEvent));
            var timer = Stopwatch.StartNew();
            while (!popup.IsEnabled)
            {
                if (timer.Elapsed > TimeSpan.FromSeconds(20)) throw new TimeoutException("Lot and wafer filter");
                Pump(); Thread.Sleep(3);
            }
            Pump();
        }
        IEnumerable<string> Runs() => ((CsvParseResult)Field(main, "_scopeData")!).Records.Select(record => record.SerialNumber);
        OpenTab(2);
        Check(List("_lotList").Items.Count == 2 && List("_waferList").Items.Count == 5 &&
            List("_lotList").SelectedItems.Count == 0 && List("_waferList").SelectedItems.Count == 0 &&
            Box("_waferList", "A / W1").IsChecked == false && Box("_waferList", "B / W1").IsChecked == false,
            "new location tab defaults to unchecked and distinguishes wafers with the same number");
        Box("_waferList", "B / W1").SetCurrentValue(ToggleButton.IsCheckedProperty, true); Pump();
        Click("应用");
        Check(Runs().SequenceEqual(new[] { "3" }), "an individually checked wafer works for records without XY");
        OpenTab(1);
        Box("_sBinList", "3").SetCurrentValue(ToggleButton.IsCheckedProperty, true); Pump();
        OpenTab(0);
        var row = ((List<SampleFilterWindow.RangeRow>)Field(popup, "_rows")!).Single();
        row.Minimum = row.Maximum = "10";
        Click("应用");
        Check(Runs().SequenceEqual(new[] { "1", "2", "3" }) &&
            ((TextBlock)main.FindName("TxtSampleFilterAction")).Text == "数据筛选（3）",
            "each tab contributes a distinct raw record to the union in the real analyzer");
        popup.Close(); Pump();
        popup = Open(); OpenTab(2);
        Check(Box("_waferList", "B / W1").IsChecked == true, "reopening restores the applied wafer selection");
        Box("_waferList", "B / W1").SetCurrentValue(ToggleButton.IsCheckedProperty, false);
        Box("_lotList", "B").SetCurrentValue(ToggleButton.IsCheckedProperty, true); Pump();
        Click("应用");
        Check(Runs().SequenceEqual(new[] { "1", "2", "3", "4", "5" }), "changing a wafer to its entire lot invalidates the cached scope");
        Click("取消数据筛选");
        Check(List("_lotList").SelectedItems.Count == 0 && List("_waferList").SelectedItems.Count == 0 &&
            List("_sBinList").SelectedItems.Count == 0 && popup.Rules.Count == 0 &&
            ReferenceEquals(original, Field(main, "_scopeData")), "clear resets all three tabs and restores the source");
        popup.Close(); Pump();
    }
    static void CheckBinClassification(string folder)
    {
        var config = ConfigLoader.Parse("[HBINClassification]\nPassHBins=4\n[SBINClassification]\nPassSBins=7\nPartialPassSBins=1");
        var records = new List<CsvRecord>
        {
            new() { Id = "a", X = 0, Y = 0, WaferId = "W", HBin = "4", SBin = "9" },
            new() { Id = "b", X = 1, Y = 0, WaferId = "W", HBin = "1", SBin = "7" },
            new() { Id = "c", X = 2, Y = 0, WaferId = "W", HBin = "5", SBin = "1" }
        };
        foreach (var record in records) record.Measurements["t"] = "1";
        var source = DatasetService.Build(new() { new() { Id = "t", Name = "T", DisplayName = "T" } }, records);
        using var lease = new DatasetLease(records);
        var hard = MapDataService.BinPages(source, false, config).Single();
        var soft = MapDataService.BinPages(source, true, config).Single();
        Check(hard.PassCount == 1 && hard.Dies.Single(d => d.HBin == "4").Color == BinPalette.PassBrush &&
            hard.Dies.Single(d => d.HBin == "1").Color != BinPalette.PassBrush,
            "HBIN Map uses configured Pass color and count, including HBIN 4");
        Check(soft.PassCount == 1 && soft.Dies.Single(d => d.SBin == "7").Color == BinPalette.PassBrush &&
            soft.Dies.Single(d => d.SBin == "1").Color != BinPalette.PassBrush,
            "SBIN Map applies its own configured Pass group");
        var fallback = new AnalyzerConfig { PassHBins = new() { "4" } };
        var legacy = MapDataService.BinPages(source, true, fallback).Single();
        Check(legacy.PassCount == 1 && legacy.Dies.Single(d => d.SBin == "9").Color == BinPalette.PassBrush,
            "SBIN Map without explicit rules follows the same record HBIN");
        var result = BinStatisticsService.Build(records, RetestMode.Last, config);
        var table = new BinStatisticsWindow(result, result);
        Check(result.PassCount == 1 && result.SBinPassCount == 1 && result.SBinPartialPassCount == 1 &&
            ((DataGrid)table.FindName("DgSbin")).Items.OfType<BinCountRow>().Single(row => row.Bin == "1").DispositionText == "PartialPass" &&
            ((DataGrid)table.FindName("DgHbin")).Items.OfType<BinCountRow>().Single(row => row.Bin == "1").DispositionText == "Fail",
            "BIN Table exposes configured dispositions separately for HBIN and SBIN");
        table.Close();
        string path = Path.Combine(folder, "configured-bin-map.xlsx");
        Wait(XlsxReportService.SaveAsync(path, source, config, new(new[] { "W" }, RetestMode.Last, null, 0, true, Array.Empty<string>()), "bin rules"));
        using var book = new XLWorkbook(path);
        var sheet = book.Worksheet("SBIN Map");
        Check(sheet.CellsUsed().Any(cell => cell.Address.ColumnNumber == 7 && cell.GetString() == "PartialPass"),
            "XLSX BIN sheet exports configured SBIN dispositions");
    }
    [STAThread] static int Main(string[] args)
    {
        try
        {
            string folder = Path.GetFullPath(args[0]); Directory.CreateDirectory(folder);
            var app = new App(); app.InitializeComponent(); app.ShutdownMode = ShutdownMode.OnExplicitShutdown;
            SynchronizationContext.SetSynchronizationContext(new DispatcherSynchronizationContext());
            var failures = new List<Exception>(); app.DispatcherUnhandledException += (_, e) => { failures.Add(e.Exception); e.Handled = true; };
            typeof(AppSettings).GetField("_userDirectory", BindingFlags.Static | BindingFlags.NonPublic)!.SetValue(null, folder);
            var config = new AnalyzerConfig { TestItemRow = 1, LowLimitRow = 2, HighLimitRow = 3, UnitRow = 4, TestNumberRow = 5,
                DataStartRow = 6, TestItemStartColumn = 8, SerialNumberColumn = 1, XColumn = 2, YColumn = 3, SiteColumn = 4, SBinColumn = 5, HBinColumn = 6, WaferColumn = 7 };
            var settings = new SettingsWindow(config, _ => { }); settings.Show(); Pump();
            ((TabControl)settings.FindName("SettingsTabs")).SelectedItem = settings.FindName("AnalysisTab"); Pump();
            var prefer = (CheckBox)settings.FindName("ChkPreferGoodRetest");
            Check(prefer.IsChecked == false, "old analyzer INI initializes good-priority checkbox off");
            prefer.IsChecked = true;
            var edited = (AnalyzerConfig)Call(settings, "ReadConfig")!;
            Check(edited.PreferGoodRetest && ((TextBox)settings.FindName("TxtConfig")).Text.Contains("[RetestSelection]"), "statistics/plot checkbox persists through actual INI editing pipeline");
            settings.Close();
            var product = new ManagedProduct { Name = "UI-good", ConfigText = ConfigLoader.Serialize(config), Flow = new() { new ProductTestStage() }, RetestMode = RetestMode.First };
            foreach (bool enabled in new[] { true, false })
            {
                var editor = new ProductEditorWindow(product);
                editor.Loaded += (_, _) => editor.Dispatcher.BeginInvoke(DispatcherPriority.ApplicationIdle, new Action(() =>
                {
                    ((CheckBox)editor.FindName("PreferGoodRetest")).IsChecked = enabled;
                    Call(editor, "PreferGoodRetest_Click", editor, new RoutedEventArgs());
                    Check(((RadioButton)editor.FindName("LastMode")).IsChecked == true && ((RadioButton)editor.FindName("FirstMode")).IsEnabled == !enabled, "product preference selects latest semantics and handles first-test comparison controls: " + enabled);
                    Call(editor, "Save_Click", editor, new RoutedEventArgs());
                }));
                Check(editor.ShowDialog() == true && editor.Product.PreferGoodRetest == enabled && editor.Product.RetestMode == RetestMode.Last, "product save persists checkbox and latest fallback: " + enabled);
                product = editor.Product;
            }
            string path = Path.Combine(folder, "retests.csv");
            File.WriteAllText(path, "SN,X,Y,Site,SBIN,HBIN,Wafer,A\nLSL,,,,,,,0\nUSL,,,,,,,10\nUnit,,,,,,,V\nTest Num,,,,,,,101\n1,0,0,1,1,1,W,1\n2,0,0,1,2,2,W,20\n3,1,0,1,2,2,W,30\n");
            var main = new MainWindow(new LayoutSettings(Path.Combine(folder, "layout.json")), new RecentFilesService(Path.Combine(folder, "recent.json")), new UiPreferencesService(Path.Combine(folder, "ui.json")));
            main.Show(); Wait(main.OpenManagedFilesAsync(new[] { new CsvImportSource(path) }, config, RetestMode.Last));
            var original = (CsvParseResult)Field(main, "_data")!;
            var grid = (DataGrid)main.FindName("DgTestItems");
            Check(grid.Items.OfType<TestItemSummary>().Single().Mean == 25, "unchecked analyzer computes latest retest measurements");
            Wait((Task)Call(main, "ApplySettings", edited)!);
            Check(ReferenceEquals(original, Field(main, "_data")) && original.Records.Count == 3 && grid.Items.OfType<TestItemSummary>().Single().Mean == 15.5, "enabling preference recalculates view without rereading or deleting raw history");
            Check(!((ListBox)main.FindName("LstRetestMode")).IsEnabled && ((ListBox)main.FindName("LstRetestMode")).SelectedIndex == (int)RetestMode.Last, "analyzer clearly shows preferred-latest mode rather than an inconsistent all-record mode");
            var pf = MapDataService.PfPages(original, original.Items[0], null, edited).Single().Dies;
            var bins = MapDataService.BinPages(original, true, edited).Single().Dies;
            Check(pf.Single(d => d.X == 0).Value == 1 && bins.Single(d => d.X == 0).SBin == "1", "PF and BIN maps follow the same good record as analyzer statistics");
            var rules = new[] { new SampleRange(original.Items[0].Id, null, 10) };
            Wait((Task)Call(main, "ApplySampleFiltersAsync", rules, SampleMatchMode.Intersection, Array.Empty<string>(), Array.Empty<SampleLocation>())!);
            Check(((CsvParseResult)Field(main, "_scopeData")!).Records.Count == 2, "sample filter admits the chip whose preferred good record satisfies the condition");
            edited.PreferGoodRetest = false; Wait((Task)Call(main, "ApplySettings", edited)!);
            Check(((CsvParseResult)Field(main, "_scopeData")!).Records.Count == 0 && ((ListBox)main.FindName("LstRetestMode")).IsEnabled, "turning preference off invalidates cached membership and restores selectable latest mode");
            Wait((Task)Call(main, "ApplySampleFiltersAsync", Array.Empty<SampleRange>(), SampleMatchMode.Intersection, Array.Empty<string>(), Array.Empty<SampleLocation>())!);
            Check(grid.Items.OfType<TestItemSummary>().Single().Mean == 25 && ReferenceEquals(original, Field(main, "_data")), "removing filter restores latest statistics over unchanged loaded data");
            CheckSBinFilter(main, grid, original);
            CheckRangeFilter(main, grid, original);
            CheckBinClassification(folder);
            edited.PreferGoodRetest = true;
            string report = Path.Combine(folder, "preferred.xlsx");
            Wait(XlsxReportService.SaveAsync(report, original, edited, new(new[] { "W" }, RetestMode.Last, null, 0, true, Array.Empty<string>()), "test"));
            using (var book = new XLWorkbook(report))
                Check(book.Worksheet(1).Cell(8, 16).GetDouble() == 15.5, "XLSX test-item summary uses preferred records consistently with GUI");
            string output = Path.Combine(folder, "preferred.csv");
            Check(IntegratedCsvService.Save(output, original, new[] { "W" }, RetestMode.Last, config: edited) == 2, "integrated CSV exports one preferred record per chip");
            var csv = DataFileService.Parse(output, edited);
            Check(csv.Items[0].Values.Any(v => v.Value == 1) && csv.Items[0].Values.All(v => v.Value != 20), "CSV export retains the preferred acquisition's actual value");
            CheckLocationFilter(main, folder, config);
            main.Close(); Pump(); Check(failures.Count == 0, "settings, filtering, export and close raise no dispatcher exceptions");
            app.Shutdown(); Console.WriteLine($"ALL {checks} GOOD RETEST UI CHECKS PASSED"); return 0;
        }
        catch (Exception ex) { Console.Error.WriteLine(ex); return 1; }
    }
