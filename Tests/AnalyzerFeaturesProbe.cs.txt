    static void Check(bool condition, string message) { if (!condition) throw new Exception(message); count++; Console.WriteLine("PASS " + message); }
    static bool Near(double? a, double b) => a.HasValue && Math.Abs(a.Value - b) <= Math.Max(1e-12, Math.Abs(b) * 1e-10);
    static object? Field(object target, string name) => target.GetType().GetField(name, BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(target);
    static void Set(object target, string name, object value) => target.GetType().GetField(name, BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(target, value);
    static object? Call(object target, string name, params object[] args) => target.GetType().GetMethod(name, BindingFlags.Instance | BindingFlags.NonPublic)!.Invoke(target, args);
    static void Pump() { var frame = new DispatcherFrame(); Dispatcher.CurrentDispatcher.BeginInvoke(DispatcherPriority.Background, new Action(() => frame.Continue = false)); Dispatcher.PushFrame(frame); }
    static void Wait(Task task) { var timer = System.Diagnostics.Stopwatch.StartNew(); while (!task.IsCompleted) { if (timer.ElapsedMilliseconds > 45000) throw new TimeoutException(); Pump(); Thread.Sleep(2); } task.GetAwaiter().GetResult(); Pump(); }
    static void Save(Window window, string name)
    {
        window.UpdateLayout(); var root = (FrameworkElement)window.Content; root.UpdateLayout();
        var drawing = new System.Windows.Media.DrawingVisual(); using (var dc = drawing.RenderOpen()) { var bounds = new Rect(0, 0, root.ActualWidth, root.ActualHeight); dc.DrawRectangle(window.Background ?? System.Windows.Media.Brushes.White, null, bounds); dc.DrawRectangle(new System.Windows.Media.VisualBrush(root), null, bounds); }
        var bitmap = new System.Windows.Media.Imaging.RenderTargetBitmap((int)root.ActualWidth, (int)root.ActualHeight, 96, 96, System.Windows.Media.PixelFormats.Pbgra32); bitmap.Render(drawing);
        var encoder = new System.Windows.Media.Imaging.PngBitmapEncoder(); encoder.Frames.Add(System.Windows.Media.Imaging.BitmapFrame.Create(bitmap)); using var file = File.Create(P(name)); encoder.Save(file);
    }

    static string Stdf()
    {
        string path = P("synthetic-units.stdf"); using var output = File.Create(path); using var writer = new BinaryWriter(output);
        void C(BinaryWriter w, string text) { var bytes = Encoding.ASCII.GetBytes(text); w.Write((byte)bytes.Length); w.Write(bytes); }
        void R(byte type, byte sub, Action<BinaryWriter> fill) { using var buffer = new MemoryStream(); using (var w = new BinaryWriter(buffer, Encoding.ASCII, true)) fill(w); writer.Write((ushort)buffer.Length); writer.Write(type); writer.Write(sub); writer.Write(buffer.ToArray()); }
        R(0, 10, w => { w.Write((byte)2); w.Write((byte)4); });
        R(1, 10, w => { w.Write((uint)0); w.Write((uint)0); w.Write((byte)1); w.Write((byte)'P'); w.Write((byte)' '); w.Write((byte)' '); w.Write((ushort)0); w.Write((byte)' '); foreach (string v in new[] { "LOT", "PRODUCT", "NODE", "TESTER", "JOB", "REV" }) C(w, v); });
        R(2, 10, w => { w.Write((byte)1); w.Write((byte)1); w.Write((uint)0); C(w, "W-STDF"); });
        R(5, 10, w => { w.Write((byte)1); w.Write((byte)1); });
        R(15, 10, w => { w.Write((uint)101); w.Write((byte)1); w.Write((byte)1); w.Write((byte)0); w.Write((byte)0); w.Write(.002f); C(w, "Current"); C(w, ""); w.Write((byte)0); w.Write((sbyte)0); w.Write((sbyte)0); w.Write((sbyte)0); w.Write(.001f); w.Write(.003f); C(w, "A"); });
        R(5, 20, w => { w.Write((byte)1); w.Write((byte)1); w.Write((byte)0); w.Write((ushort)1); w.Write((ushort)1); w.Write((ushort)1); w.Write((short)2); w.Write((short)3); w.Write((uint)0); C(w, "STDF-1"); });
        R(2, 20, w => { w.Write((byte)1); w.Write((byte)1); for (int i = 0; i < 6; i++) w.Write((uint)0); C(w, "W-STDF"); }); R(1, 20, w => { w.Write((uint)0); w.Write((byte)' '); }); return path;
    }

        var waferWindow = new WaferTableWindow(WaferStatisticsService.Build(records, RetestMode.First, labels), wafers); waferWindow.Show(); Pump();
        Check(((DataGrid)waferWindow.FindName("DgWafers")).Columns.Count == 4 && ((string)Call(waferWindow, "BuildClipboardText")!).Contains("开路=Open"), "Wafer Table adds description without shifting numeric wafer columns"); Save(waferWindow, "wafer-descriptions.png"); waferWindow.Close();
        string path = P("fail-chips.xlsx"); var result = FailChipXlsxService.Save(path, data, labels, new[] { "W" }, RetestMode.All);
        Check(result.FailChips == 3 && result.ReferenceSerial == "g2" && result.RetestGoodChips == 1 && result.Records == 7, "Fail export writes recovered good chips separately and preserves closest good-only reference");
        using (var doc = DocumentFormat.OpenXml.Packaging.SpreadsheetDocument.Open(path, false))
        {
            var errors = new DocumentFormat.OpenXml.Validation.OpenXmlValidator().Validate(doc).ToArray();
            foreach (var error in errors) Console.WriteLine(error.Description + " at " + error.Path?.XPath);
            Check(errors.Length == 0, "streamed XLSX conforms to the Open XML schema");
        }
        using (var workbook = new XLWorkbook(path))
        {
            Check(workbook.Worksheets.Count == 2 && workbook.Worksheet(2).Name == "复测良品芯片" &&
                workbook.Worksheet(2).RowsUsed().Skip(7).Select(r => r.Cell(12).GetString()).SequenceEqual(new[] { "other-first", "other-last" }), "recovered good chip history is on the second sheet");
            var sheet = workbook.Worksheet(1); var rows = sheet.RowsUsed().Where(r => records.Any(s => s.Id == r.Cell(12).GetString())).ToDictionary(r => r.Cell(12).GetString());
            Check(rows.Count == 5 && rows["f-last"].RowNumber() == rows["f-first"].RowNumber() + 1 && !rows.ContainsKey("other-first"), "first and retest records are adjacent inside chip blocks and recovered good chips are excluded");
            Check(rows["f-first"].Cell(13).Style.Fill.BackgroundColor.Color.ToArgb() == System.Drawing.Color.FromArgb(255, 169, 209, 142).ToArgb() &&
                rows["f-last"].Cell(13).Style.Fill.BackgroundColor.Color.ToArgb() == System.Drawing.Color.FromArgb(255, 242, 160, 175).ToArgb(), "measurement cells are green on pass and pink on fail");
            Check(Near(rows["g2"].Cell(13).GetDouble(), 2) && sheet.Cell(3, 13).GetString() == "mA" && Near(sheet.Cell(5, 13).GetDouble(), 5), "Fail XLSX applies STDF display units to values and limits");
            Check(rows["invalid"].Cell(13).GetString() == "bad" && rows["invalid"].Cell(14).IsEmpty(), "invalid text and missing values remain distinguishable in export");
        }
        var noGood = new CsvParseResult { Items = data.Items, Records = records.Where(r => r.X == 3 || r.X == 5).ToList() };
        var missingRef = FailChipXlsxService.Save(P("no-reference.xlsx"), noGood, labels, new[] { "W" }, RetestMode.All);
        Check(missingRef.ReferenceSerial == null && missingRef.FailChips == 2, "all-fail population exports without inventing a good reference");
        string untouched = P("keep.xlsx"); File.WriteAllText(untouched, "keep"); bool noFails = false;
        try { FailChipXlsxService.Save(untouched, new CsvParseResult { Items = data.Items, Records = records.Where(r => r.X < 3).ToList() }, labels, new[] { "W" }, RetestMode.All); }
        catch (InvalidDataException) { noFails = true; } Check(noFails && File.ReadAllText(untouched) == "keep", "no-fail selection leaves an existing destination untouched");
        var lastOnly = FailChipXlsxService.Save(P("last-only.xlsx"), data, labels, new[] { "W" }, RetestMode.Last);
        Check(lastOnly.Records == 5 && data.Records.Count == 9 && Near(first.Values[1].Value, .002), "export honors retest selection and does not alter source measurements");
        var onlyRetests = FailChipXlsxService.Save(P("only-retests.xlsx"), data, labels, new[] { "W" }, RetestMode.OnlyRetests);
        Check(onlyRetests.FailChips == 1 && onlyRetests.Records == 3 && onlyRetests.ReferenceSerial == "other-last", "only-retest export retains selected fail history and a valid retested reference");
        var ftRecords = records.Where(r => r.X == 3).Select((r, i) => new CsvRecord { Id = r.Id, WaferId = i == 0 ? "W" : "W2", SerialNumber = r.SerialNumber,
            X = i * 20, Y = 0, Lot = "L", IsFtIdentity = true, RetestUuid = "UUID", HBin = r.HBin, Measurements = r.Measurements }).ToList();
        var ft = FailChipXlsxService.Save(P("ft-history.xlsx"), new CsvParseResult { Items = data.Items, Records = ftRecords }, labels, new[] { "W", "W2" }, RetestMode.All);
        Check(ft.FailChips == 1 && ft.Records == 2, "FT UUID groups one chip across coordinate and wafer changes");
        string clippedPath = P("fail-outliers.xlsx"); FailChipXlsxService.Save(clippedPath, data, labels, new[] { "W" }, RetestMode.All, 1);
        using (var clipped = new XLWorkbook(clippedPath))
        {
            var r = clipped.Worksheet(1).RowsUsed().Single(r => r.Cell(12).GetString() == "f-last");
            Check(r.Cell(13).IsEmpty() && r.Cell(13).Style.Fill.BackgroundColor.Color.ToArgb() == System.Drawing.Color.FromArgb(255, 217, 222, 227).ToArgb(), "optional outlier clipping blanks only the affected measurement with a neutral style");
        }
        var cancellation = new CancellationTokenSource(); cancellation.Cancel(); bool stopped = false;
        using (new AnalysisOperation(cancellation.Token)) try { FailChipXlsxService.Save(P("cancelled.xlsx"), data, labels, new[] { "W" }, RetestMode.All); } catch (OperationCanceledException) { stopped = true; }
        Check(stopped && !File.Exists(P("cancelled.xlsx")), "cancelled Fail export leaves no partial destination");
    }
