    static int checks;
    static string folder = "";
    static string P(string name) => Path.Combine(folder, name);
    static void Check(bool value, string message)
    { if (!value) throw new Exception(message); checks++; Console.WriteLine("PASS " + message); }
    static void Reject<T>(Action action, string message) where T : Exception
    { try { action(); } catch (T) { Check(true, message); return; } throw new Exception(message); }
    static long OwnNumbers(TestValueColumn c) => new[] { "_numbers", "_singleNumbers" }.Sum(name =>
        typeof(TestValueColumn).GetField(name, BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(c) is Array a ? Buffer.ByteLength(a) : 0L);
    static AnalyzerConfig Config() => new()
    {
        TestItemRow = 1, LowLimitRow = 2, HighLimitRow = 3, UnitRow = 4, TestNumberRow = 5,
        SBinRow = 0, HBinRow = 0, DataStartRow = 6, TestItemStartColumn = 8,
        SerialNumberColumn = 1, XColumn = 2, YColumn = 3, SiteColumn = 4, SBinColumn = 5, HBinColumn = 6, WaferColumn = 7
    };
    static string WriteCsv(string name, string rows, Encoding? encoding = null)
    {
        string path = P(name);
        File.WriteAllText(path, "SN,X,Y,Site,SBIN,HBIN,Wafer,UUID,Value\nLSL,,,,,,,0,0\nUSL,,,,,,,100000,2\nUnit,,,,,,,V,A\nTest Num,,,,,,,101,102\n" + rows, encoding ?? new UTF8Encoding(false));
        return path;
    }
    static CsvRecord Record(string id, int x, params (string Key, string Value)[] cells) => new()
    {
        Id = id, SerialNumber = id, X = x, Y = 0, Site = "1", WaferId = "W", HBin = "1",
        Measurements = cells.ToDictionary(p => p.Key, p => p.Value)
    };
    static CsvParseResult Fixture()
    {
        var items = new[] { "a", "b", "c" }.Select(id => new TestItemData { Id = id, Name = id, TestNumber = id, LowLimit = 0, HighLimit = 2 }).ToList();
        return DatasetService.Build(items, new()
        {
            Record("r0", 0, ("a", "1.0000000000000002"), ("b", "bad")),
            Record("r1", 0, ("a", "2"), ("c", "")),
            Record("r2", 0, ("a", "3"), ("b", "4")),
            Record("r3", 1, ("a", "5"), ("b", StdfParser.NotApplicableValue)),
            Record("r4", 2)
        });
    }
    static string Cells(IEnumerable<CsvRecord> records) => string.Join(";", records.Select(r =>
        r.Id + ":" + string.Join(',', r.Measurements.Keys.Order(StringComparer.Ordinal))));
    static void SameView(CsvParseResult view, string name)
    {
        var rebuilt = DatasetService.Build(view.Items.Select(DatasetService.CloneItem).ToList(), view.Records);
        Check(view.Items.Zip(rebuilt.Items).All(p => p.First.Values.Select(v => (v.RecordId, v.Value, v.RawText))
            .SequenceEqual(p.Second.Values.Select(v => (v.RecordId, v.Value, v.RawText))) &&
            p.First.Errors.Select(r => r.Id).SequenceEqual(p.Second.Errors.Select(r => r.Id))), name + " values/errors/raw text match materialized data");
        Check(view.Items.All(i => OwnNumbers(i.Values) == 0), name + " has no copied numeric arrays");
        Check(view.Items.Zip(rebuilt.Items).All(p => Enumerable.Range(0, view.Records.Count).All(row =>
            p.First.Values.GetSourceRowValue(row).Equals(p.Second.Values.GetSourceRowValue(row)))), name + " source row lookup includes absent/invalid cells");
        foreach (var pair in view.Items.Zip(rebuilt.Items))
        {
            var a = StatisticsService.CalculateSummaryMetrics(pair.First.Values, 0, 2, true, 6);
            var b = StatisticsService.CalculateSummaryMetrics(pair.Second.Values, 0, 2, true, 6);
            Check(typeof(StatisticsResult).GetProperties().All(p => Equals(p.GetValue(a.Stats), p.GetValue(b.Stats))) &&
                a.Median == b.Median && a.Robust == b.Robust, name + " exact statistics " + pair.First.Id);
        }
        var snapshot = SimulationSnapshot.Build(view, Config(), RetestMode.Last, default);
        var expected = SimulationSnapshot.Build(rebuilt, Config(), RetestMode.Last, default);
        Check(snapshot.Count == expected.Count && snapshot.States.Zip(expected.States).All(p => p.First.SequenceEqual(p.Second)) &&
            Enumerable.Range(0, snapshot.Count).All(chip => Enumerable.Range(0, view.Items.Count).All(item =>
                snapshot.Value(chip, item).Equals(expected.Value(chip, item)))), name + " simulation mapping");
    }
    static void Views()
    {
        using var operation = new AnalysisOperation();
        var source = Fixture();
        var expected = new Dictionary<RetestMode, string>
        {
            [RetestMode.First] = "r0:a,b;r1:c;r3:a,b;r4:",
            [RetestMode.Last] = "r1:c;r2:a,b;r3:a,b;r4:",
            [RetestMode.OnlyRetests] = "r1:a;r2:a,b"
        };
        foreach (var mode in expected.Keys)
        {
            var selected = DatasetService.SelectRetests(source, mode);
            Check(Cells(selected.Records) == expected[mode], mode + " selects each test independently including invalid values");
            SameView(selected, mode.ToString());
            var path = P(mode + "-integrated.csv");
            IntegratedCsvService.Save(path, selected);
            var loaded = IntegratedCsvService.Load(path);
            Check(loaded.Records.Select(r => r.Measurements.Values.ToArray()).Zip(selected.Records)
                .All(p => p.First.SequenceEqual(p.Second.Measurements.Values)), mode + " integrated export preserves selected raw cells");
        }
        var filtered = SampleFilterService.Apply(source, new[] { new SampleRange("a", 1, 3) }, RetestMode.All);
        Check(filtered.Records.Count == 3, "range filter selects original rows");
        SameView(filtered, "filter");
        SameView(DatasetService.SelectRetests(filtered, RetestMode.Last), "nested retest");
        foreach (var mode in new[] { RetestMode.First, RetestMode.Last, RetestMode.OnlyRetests })
            SameView(SampleFilterService.Apply(source, new[] { new SampleRange("a", 1, 4) }, mode), "filter in " + mode);
        var empty = SampleFilterService.Apply(source, new[] { new SampleRange("a", 100, null) }, RetestMode.All);
        Check(empty.Records.Count == 0 && empty.Items.All(i => i.Values.Count == 0 && i.Errors.Count == 0), "empty view is valid");
        var view = DatasetService.SelectRetests(source, RetestMode.First);
        double original = view.Items[0].Values[0].Value;
        source.Items[0].Values[0] = new(source.Records[0], 99);
        Check(view.Items[0].Values[0].Value == original, "source edit detaches from existing view");
        view.Items[0].Values[0] = new(view.Records[0], 88);
        Check(source.Items[0].Values[0].Value == 99, "view edit detaches from source");
        source.Items[0].Values.Clear();
        Check(view.Items[0].Values.Count > 0 && view.Items[0].Values[0].Value == 88, "source clear does not invalidate detached view");
        var floats = DatasetService.Build(new() { new() { Id = "f", Name = "f", IsStdfMeasurement = true } },
            new() { Record("s0", 0, ("f", "1.25")), Record("s1", 0, ("f", "2.5")), Record("s2", 2), Record("s3", 3), Record("s4", 4) });
        SameView(DatasetService.SelectRetests(floats, RetestMode.Last), "sparse float");
        using var cancellation = new CancellationTokenSource(); cancellation.Cancel();
        using var canceled = new AnalysisOperation(cancellation.Token);
        Reject<OperationCanceledException>(() => DatasetService.SelectRetests(source, RetestMode.Last), "view construction honors cancellation");
    }
    static void SelectionMatrix()
    {
        using var ownership = new AnalysisOperation();
        var random = new Random(7241);
        foreach (bool ft in new[] { false, true })
        {
            var items = Enumerable.Range(0, 7).Select(i => new TestItemData { Id = "t" + i, Name = "t" + i }).ToList();
            var records = Enumerable.Range(0, 60).Select(row =>
            {
                var record = Record("m" + row, row % 13);
                record.IsFtIdentity = ft;
                record.RetestUuid = row % 9 == 0 ? "" : "uuid" + row % 13;
                record.Lot = "L" + row % 2;
                foreach (var item in items)
                    if (random.Next(3) != 0) record.Measurements.Add(item.Id, random.Next(4) == 0 ? "bad" : row.ToString(CultureInfo.InvariantCulture));
                return record;
            }).ToList();
            var data = DatasetService.Build(items, records);
            foreach (var mode in new[] { RetestMode.First, RetestMode.Last, RetestMode.OnlyRetests })
            {
                // Independent per-cell oracle: gather the execution history of each
                // measurement, then select its first/last/all-subsequent occurrences.
                var expected = new Dictionary<string, HashSet<string>>();
                foreach (var chip in records.GroupBy(r => ft
                    ? (r.RetestUuid == "" ? "ROW:" + r.Id : r.Lot + ":" + r.RetestUuid) : "XY:" + r.X))
                {
                    var history = chip.ToArray();
                    if (history.Length == 1 && mode != RetestMode.OnlyRetests)
                        expected[history[0].Id] = history[0].Measurements.Keys.ToHashSet();
                    foreach (var item in items)
                    {
                        var executions = history.Where(r => r.Measurements.ContainsKey(item.Id)).ToArray();
                        var chosen = mode == RetestMode.First ? executions.Take(1) : mode == RetestMode.Last ? executions.TakeLast(1) : executions.Skip(1);
                        foreach (var record in chosen)
                        {
                            if (!expected.TryGetValue(record.Id, out var keys)) expected[record.Id] = keys = new();
                            keys.Add(item.Id);
                        }
                    }
                }
                var selected = DatasetService.SelectRetests(data, mode);
                Check(selected.Records.Select(r => r.Id).SequenceEqual(records.Where(r => expected.ContainsKey(r.Id)).Select(r => r.Id)) &&
                    selected.Records.All(r => expected[r.Id].SetEquals(r.Measurements.Keys)), $"mixed partial histories {ft}/{mode}");
                SameView(selected, $"mixed {ft}/{mode}");
            }
        }
        var source = Fixture();
        var sourceLease = new DatasetLease(source.Records);
        var view = DatasetService.SelectRetests(source, RetestMode.Last);
        using var viewLease = new DatasetLease(view.Records);
        sourceLease.Dispose();
        Check(view.Items[0].Values[0].RawText == "3", "view lease keeps raw snapshot alive after source lease release");
    }
    static CsvParseResult SameProjection(string path, AnalyzerConfig config, string label, StdfImportMode mode = StdfImportMode.Standard)
    {
        using var operation = new AnalysisOperation();
        var full = DataFileService.Parse(path, config, stdfMode: mode);
        var projected = DataFileService.ReadForYield(path, config, default, mode);
        Check(projected.IsYieldProjection && projected.Items.All(i => i.Values.Count == 0 && OwnNumbers(i.Values) == 0) &&
            projected.Records.All(r => !r.Measurements.IsOnDisk), label + " allocates no full numeric columns or raw store");
        object Identity(CsvRecord r) => (r.SerialNumber, r.OriginalSerialNumber, r.WaferId, r.X, r.Y, r.Site, r.HBin, r.SBin,
            r.Product, r.Program, r.Lot, r.StdfTestTime, r.IsFtIdentity, r.RetestUuid, r.UuidTestNumbers);
        Check(full.Records.Count == projected.Records.Count && full.Records.Zip(projected.Records).All(p =>
            Equals(Identity(p.First), Identity(p.Second)) && p.Second.HasNonblankMeasurements == p.First.Measurements.Any(v => !string.IsNullOrWhiteSpace(v.Value))),
            label + " preserves identities, bins, times, UUID and blank-row semantics");
        var numbers = config.TestMode == TestAnalysisMode.FT ? config.UuidTestNumbers.ToHashSet() : new HashSet<string>();
        Check(projected.Records.All(r => r.Measurements.Keys.All(key => numbers.Contains(projected.Items.Single(i => i.Id == key).TestNumber))),
            label + " retains only configured UUID measurements");
        return projected;
    }
    static void Projections()
    {
        var config = Config();
        string rows = "DUP,0,0,1,1,1,W1,00123,1.5\nDUP,0,0,1,2,2,W1,00123,2.5\n,1,0,2,1,1,W1,0x00AF,\"bad\ntext\"\n,,,,,,,,\n,,,,,,,,3\n";
        var path = WriteCsv("projection.csv", rows);
        SameProjection(path, config, "CSV CP");
        config.TestMode = TestAnalysisMode.FT; config.UuidTestNumbers = new() { "101" };
        var ft = SameProjection(path, config, "CSV FT");
        Check(ft.Records[0].RetestUuid == "00123" && ft.Records[2].RetestUuid == "0x00AF", "UUID keeps leading zeroes and hex text");
        SameProjection(WriteCsv("unicode.csv", rows, Encoding.Unicode), config, "UTF16 FT");
        var repeated = WriteCsv("repeated.csv", rows);
        File.AppendAllText(repeated, File.ReadAllText(repeated));
        SameProjection(repeated, config, "repeated headers and multiline cells");
        using (var operation = new AnalysisOperation())
        {
            var data = DataFileService.Parse(path, config);
            var integrated = P("projection-integrated.csv");
            IntegratedCsvService.Save(integrated, data);
            SameProjection(integrated, config, "integrated FT");
            // A partial retest may carry its derived UUID but none of the source UUID cells.
            data.Records[0].Measurements = data.Records[0].Measurements.SelectKeys(key => key != data.Items[0].Id);
            IntegratedCsvService.Save(integrated, data);
            var partial = SameProjection(integrated, config, "integrated partial UUID");
            Check(partial.Records[0].RetestUuid == "00123", "derived UUID survives partial integrated export");
        }
        var invalid = WriteCsv("ambiguous.csv", rows);
        File.WriteAllText(invalid, File.ReadAllText(invalid).Replace("101,102", "101,101"));
        Reject<InvalidDataException>(() => DataFileService.ReadForYield(invalid, config, default), "ambiguous UUID definitions still rejected");
        config.UuidTestNumbers = new() { "999" };
        var missing = SameProjection(path, config, "missing UUID test");
        Check(missing.Records.All(r => r.RetestUuid == ""), "missing UUID cannot invent identity");
        foreach (bool big in new[] { false, true })
        {
            var stdf = StdfFixture(big);
            var cp = Config(); SameProjection(stdf, cp, "STDF CP " + big);
            var sf = Config(); sf.TestMode = TestAnalysisMode.FT; sf.UuidTestNumbers = new() { "900" };
            SameProjection(stdf, sf, "STDF FT " + big);
            SameProjection(stdf, sf, "STDF 93k " + big, StdfImportMode.Advantest93K);
            using (var input = File.OpenRead(stdf))
            using (var output = File.Create(stdf + ".gz"))
            using (var gzip = new GZipStream(output, CompressionLevel.Fastest)) input.CopyTo(gzip);
            SameProjection(stdf + ".gz", sf, "compressed STDF " + big);
            var broken = P("truncated.stdf"); File.WriteAllBytes(broken, File.ReadAllBytes(stdf)[..^5]);
            Reject<InvalidDataException>(() => DataFileService.ReadForYield(broken, cp, default), "truncated STDF rejected " + big);
        }
        using var cts = new CancellationTokenSource(); cts.Cancel();
        Reject<OperationCanceledException>(() => DataFileService.ReadForYield(path, config, cts.Token), "yield read honors cancellation");
    }
    static string StdfFixture(bool big, uint? firstStart = null, uint? secondStart = null)
    {
        var path = P(big ? "big.stdf" : "little.stdf");
        using var output = File.Create(path); using var writer = new BinaryWriter(output);
        void U2(BinaryWriter w, ushort n) { Span<byte> b = stackalloc byte[2]; if (big) BinaryPrimitives.WriteUInt16BigEndian(b, n); else BinaryPrimitives.WriteUInt16LittleEndian(b, n); w.Write(b); }
        void U4(BinaryWriter w, uint n) { Span<byte> b = stackalloc byte[4]; if (big) BinaryPrimitives.WriteUInt32BigEndian(b, n); else BinaryPrimitives.WriteUInt32LittleEndian(b, n); w.Write(b); }
        void C(BinaryWriter w, string text) { var b = Encoding.ASCII.GetBytes(text); w.Write((byte)b.Length); w.Write(b); }
        void R(byte type, byte sub, Action<BinaryWriter> fill)
        { using var buffer = new MemoryStream(); using (var w = new BinaryWriter(buffer, Encoding.ASCII, true)) fill(w); U2(writer, (ushort)buffer.Length); writer.Write(type); writer.Write(sub); writer.Write(buffer.ToArray()); }
        R(0, 10, w => { w.Write((byte)(big ? 1 : 2)); w.Write((byte)4); });
        R(1, 10, w => { U4(w, 0); U4(w, 10); w.Write((byte)1); w.Write((byte)'P'); w.Write((byte)' '); w.Write((byte)' '); U2(w, 0); w.Write((byte)' '); foreach (var s in new[] { "LOT", "PRODUCT", "NODE", "TESTER", "JOB", "REV" }) C(w, s); });
        for (int wafer = 0; wafer < 2; wafer++)
        {
            R(2, 10, w => { w.Write((byte)1); w.Write((byte)1); U4(w, wafer == 0 ? firstStart ?? 20U : secondStart ?? 30U); C(w, "W" + wafer); });
            R(5, 10, w => { w.Write((byte)1); w.Write((byte)1); });
            R(15, 10, w => { U4(w, 900); w.Write((byte)1); w.Write((byte)1); w.Write((byte)0); w.Write((byte)0); U4(w, BitConverter.SingleToUInt32Bits(1.25f + wafer)); C(w, "UUID"); C(w, ""); });
            R(15, 15, w => { U4(w, 901); w.Write((byte)1); w.Write((byte)1); w.Write((byte)0); w.Write((byte)0); U2(w, 0); U2(w, 2); U4(w, BitConverter.SingleToUInt32Bits(2)); U4(w, BitConverter.SingleToUInt32Bits(3)); });
            R(15, 20, w => { U4(w, 902); w.Write((byte)1); w.Write((byte)1); w.Write((byte)0); });
            R(5, 20, w => { w.Write((byte)1); w.Write((byte)1); w.Write((byte)0); U2(w, 4); U2(w, (ushort)(wafer + 1)); U2(w, 1); U2(w, 2); U2(w, 3); U4(w, 0); C(w, "S" + wafer); });
            R(2, 20, w => { w.Write((byte)1); w.Write((byte)1); for (int i = 0; i < 6; i++) U4(w, 30); C(w, "W" + wafer); });
        }
        R(1, 20, w => { U4(w, 40); w.Write((byte)' '); });
        return path;
    }
    static void ProductYield()
    {
        var config = Config();
        var cold = new ProductTestStage { Id = "cold", Name = "cold", Kind = "CP", Required = true };
        var hot = new ProductTestStage { Id = "hot", Name = "hot", Kind = "CP", Required = true };
        var a = WriteCsv("cold.csv", "A,0,0,1,1,1,W,001,1\nB,1,0,1,1,1,W,002,1\nA2,0,0,1,2,2,W,001,3\n");
        var b = WriteCsv("hot.csv", "A,0,0,1,1,1,W,001,1\n");
        var product = new ManagedProduct { Name = "test", ConfigText = ConfigLoader.Serialize(config), Flow = new() { cold, hot },
            Files = new() { new() { StageId = cold.Id, FilePath = a, Lot = "L" }, new() { StageId = hot.Id, FilePath = b, Lot = "L" } } };
        foreach (var mode in new[] { RetestMode.First, RetestMode.Last })
        {
            product.RetestMode = mode;
            var result = ProductYieldService.Calculate(product);
            var combined = result.Yields.Single(r => r.IsCombined);
            Check(result.Files.All(f => f.Error == "") && combined.Total == 2 && combined.Good == (mode == RetestMode.First ? 1 : 0) && combined.MissingTemperature == 1,
                "CP product yield keeps per-chip multi-temperature and " + mode + " semantics");
        }
        cold.Kind = hot.Kind = "FT"; product.UuidTestNumbers = "101";
        var ft = ProductYieldService.Calculate(product);
        Check(ft.Files.All(f => f.Error == "") && ft.Yields.Single(r => r.IsCombined).Total == 2, "FT product yield matches UUID across stages");
        cold.Kind = hot.Kind = "CP";
        product.Files.Add(new() { StageId = cold.Id, FilePath = P("missing.csv"), Lot = "L" });
        var failed = ProductYieldService.Calculate(product);
        Check(failed.Files.Last().Error.Length > 0 && failed.Yields.Single(r => r.IsCombined).Total == 2, "failed file contributes no partial population");
        using (var operation = new AnalysisOperation())
        {
            var transaction = WriteCsv("transaction.csv", "NEW,10,0,1,1,1,W,003,1\n");
            var broken = P("transaction-integrated.csv");
            IntegratedCsvService.Save(broken, DataFileService.Parse(transaction, config));
            File.AppendAllText(broken, "BROKEN\n");
            product.Files.Add(new() { StageId = cold.Id, FilePath = broken, Lot = "L" });
            var incomplete = ProductYieldService.Calculate(product);
            Check(incomplete.Files.Last().Error.Length > 0 && incomplete.Yields.Single(r => r.IsCombined).Total == 2,
                "malformed file after valid rows rolls back the entire file");
        }
        product.Files = new() { new() { StageId = cold.Id, FilePath = StdfFixture(false), Lot = "L", WaferOverride = "W" } };
        var multi = ProductYieldService.Calculate(product);
        Check(multi.Files.Single().Error.Contains("多个晶圆") && multi.Yields.Count == 0, "wafer override still rejects multi-wafer files");
    }
    static void YieldRowSources()
    {
        var config = Config();
        var cold = new ProductTestStage { Id = "row-cold", Name = "cold", Kind = "CP", Required = true };
        var hot = new ProductTestStage { Id = "row-hot", Name = "hot", Kind = "CP", Required = true };
        ManagedDataFile File(string name, string stage, string lot, string wafer, int x, DateTime? time) => new()
        {
            Id = name, StageId = stage, Lot = lot, TestTime = time,
            FilePath = WriteCsv(name + ".csv", $"SN,{x},0,1,1,1,{wafer},001,1\n")
        };
        var old = File("row-old", cold.Id, "L1", "W1", 0, new(2026, 1, 10));
        var latest = File("row-latest", cold.Id, "L1", "W1", 0, new(2026, 1, 20));
        var otherWafer = File("row-wafer", cold.Id, "L1", "W2", 1, new(2026, 1, 15));
        var otherLot = File("row-lot", cold.Id, "L2", "W1", 0, new(2026, 1, 25));
        var hotFile = File("row-hot-file", hot.Id, "L1", "W1", 0, new(2026, 2, 15));
        var undated = File("row-undated", cold.Id, "L1", "W1", 9, null);
        var failed = new ManagedDataFile { Id = "row-failed", StageId = cold.Id, Lot = "L1", FilePath = P("does-not-exist.csv") };
        var product = new ManagedProduct { Name = "row sources", ConfigText = ConfigLoader.Serialize(config), Flow = new() { cold, hot },
            Files = new() { old, latest, otherWafer, otherLot, hotFile, undated, failed } };
        product.Snapshot = ProductYieldService.Calculate(product);
        void Sources(ManagedYieldRow row, string message, params ManagedDataFile[] expected)
        {
            Check(row.SourceFileIds.SetEquals(expected.Select(f => f.Id)), message + " exact source IDs");
            Check(ManagedAnalysisService.YieldFiles(product, row).Select(f => f.Id)
                .SequenceEqual(product.Files.Where(f => expected.Contains(f)).Select(f => f.Id)), message + " registration order");
        }
        var wafer = ManagementTrendService.DetailYieldRows(product).Single(r => r.StageId == cold.Id && r.Lot == "L1" && r.Wafer == "W1");
        Sources(wafer, "wafer row includes overwritten retest history", old, latest, undated);
        Sources(ManagementTrendService.DetailYieldRows(product, period: ManagementPeriod.Lot).Single(r => r.StageId == cold.Id && r.Lot == "L1"),
            "lot aggregation", old, latest, otherWafer, undated);
        var january = ManagementTrendService.DetailYieldRows(product, period: ManagementPeriod.Month).Single(r => r.StageId == cold.Id && r.Period == "2026-01");
        Sources(january, "month aggregation across lots", old, latest, otherWafer, otherLot);
        Sources(ManagementTrendService.DetailYieldRows(product, "W2", ManagementPeriod.Month).Single(r => r.StageId == cold.Id),
            "wafer filter before monthly aggregation", otherWafer);
        Sources(ManagementTrendService.DetailYieldRows(product, period: ManagementPeriod.Year).Single(r => r.StageId == cold.Id && r.Period == "2026"),
            "year aggregation", old, latest, otherWafer, otherLot);
        Sources(ManagementTrendService.DetailYieldRows(product, period: ManagementPeriod.Month).Single(r => r.StageId == cold.Id && r.Period == ""),
            "unknown time is isolated", undated);
        var february = ManagementTrendService.DetailYieldRows(product, period: ManagementPeriod.Month).Single(r => r.IsCombined && r.Period == "2026-02");
        Sources(february, "combined February retains January temperature sources", old, latest, hotFile);
        var batches = ManagedAnalysisService.YieldSources(product, february);
        Check(batches.Length == 2 && batches[0].Select(s => s.Path).SequenceEqual(new[] { old.FilePath, latest.FilePath }) &&
            batches[1].Single().Path == hotFile.FilePath && batches[0][1].Options?.AutoMatch == true,
            "combined row opens one merged batch per step");
        var monthSources = ManagedAnalysisService.YieldSources(product, january);
        Check(monthSources.Length == 1 && monthSources[0].Length == 4 && monthSources[0].All(s => s.ShowLotInWafer),
            "same-step month merges all lots with separate wafer identity");
        Check(ProductCatalogStore.Clone(february).SourceFileIds.SetEquals(february.SourceFileIds), "source links survive catalog serialization");
        product.RetestMode = RetestMode.First;
        product.Snapshot = ProductYieldService.Calculate(product);
        Sources(ManagementTrendService.DetailYieldRows(product).Single(r => r.StageId == cold.Id && r.Lot == "L1" && r.Wafer == "W1"),
            "first-test mode keeps all matching source files", old, latest, undated);
        uint Stamp(DateTime date) => (uint)new DateTimeOffset(DateTime.SpecifyKind(date, DateTimeKind.Utc)).ToUnixTimeSeconds();
        var multiPath = StdfFixture(false, Stamp(new(2026, 1, 15)), Stamp(new(2026, 2, 15)));
        var metadata = StdfMetadataService.Read(multiPath);
        var multiFile = new ManagedDataFile { Id = "row-multi", StageId = cold.Id, FilePath = multiPath, Lot = "MULTI",
            UseStdfTestTime = true, TestTime = metadata.TestTime, DetectedStdfTestTime = metadata.TestTime };
        product.Files = new() { multiFile };
        product.Snapshot = ProductYieldService.Calculate(product);
        Sources(ManagementTrendService.DetailYieldRows(product, "W0", ManagementPeriod.Month).Single(r => r.StageId == cold.Id && r.Period == "2026-01"),
            "multi-wafer STDF actual wafer timestamp", multiFile);
        Check(!ManagementTrendService.DetailYieldRows(product, "W0", ManagementPeriod.Month).Any(r => r.StageId == cold.Id && r.Period == "2026-02"),
            "multi-wafer timestamp does not leak into another wafer month");
        var specified = new ManagedDataFile { Id = "row-specified", StageId = cold.Id, Lot = "S", WaferOverride = "W", UseSpecifiedYield = true,
            SpecifiedKind = "CP", SpecifiedTotal = 10, SpecifiedGood = 9, TestTime = new(2026, 3, 15) };
        product.Files = new() { specified };
        product.Snapshot = ProductYieldService.Calculate(product);
        var manual = ManagementTrendService.DetailYieldRows(product).Single(r => r.StageId == cold.Id);
        Check(manual.SourceFileIds.SetEquals(new[] { specified.Id }), "specified row retains its registration link");
        Reject<InvalidDataException>(() => ManagedAnalysisService.YieldFiles(product, manual), "specified row without source reports no original file");
        var unknown = product.Snapshot.Yields.Single(r => r.IsCombined);
        Check(unknown.SourceFileIds.SetEquals(new[] { specified.Id }), "unavailable combined row retains underlying source registration");
        product.Snapshot.DefinitionSignature = "old-version";
        Check(!ProductYieldService.IsCurrent(product), "old snapshots require recomputation before file navigation");
    }
    static void ScatterMarking()
    {
        Check(System.Runtime.CompilerServices.Unsafe.SizeOf<ScatterPoint>() == 24, "scatter identity index fits existing 24-byte point layout");
        var records = new[]
        {
            new CsvRecord { Id = "a", Lot = "L1", WaferId = "W1", X = 1, Y = 2, Site = "2", SerialNumber = "2" },
            new CsvRecord { Id = "b", Lot = "L1", WaferId = "W1", X = 3, Y = 2, Site = "1", SerialNumber = "1" },
            new CsvRecord { Id = "c", Lot = "L1", WaferId = "W2", X = 1, Y = 2, Site = "1", SerialNumber = "3" },
            new CsvRecord { Id = "d", Lot = "L2", WaferId = "W1", X = 1, Y = 2, Site = "2", SerialNumber = "4" },
            new CsvRecord { Id = "retest", Lot = "L1", WaferId = "W1", X = 1, Y = 2, Site = "3", SerialNumber = "5" }
        };
        var values = records.Select((r, i) => new TestValue(r, i == 1 ? double.NaN : i + 10)).ToArray();
        var marks = new ScatterChipMarks();
        var first = ScatterSeries.Build(values, 0, 11, trackChips: true);
        Check(!marks.AddRectangle(first, double.NaN, 5, 0, 20), "nonfinite selection rejected");
        Check(!marks.AddRectangle(ScatterSeries.Build(values, null, null), 1, 5, 0, 20), "noninteractive series does not retain chip identities");
        Check(marks.AddRectangle(first, 1, 1, 10, 10) && marks.Count == 1, "inclusive rectangle selects original physical chip");
        Check(marks.CreateMask(first).SequenceEqual(new[] { true, false, false, true }), "retest is marked while other wafer/lot coordinates stay separate");
        Check(!marks.AddRectangle(first, 1, 1, 10, 10) && marks.Count == 1, "repeated marking is idempotent");
        Check(!marks.AddRectangle(first, 2, 2, -100, 100), "NaN gap retains its serial index without selecting adjacent chip");
        Check(!marks.AddRectangle(first, 1.01, 1.99, 0, 20), "selection uses point centers and precise bounds");
        var nextValues = records.Reverse().Select((r, i) => new TestValue(r, 100 + i)).ToArray();
        foreach (var order in Enum.GetValues<ScatterOrder>())
        {
            var next = ScatterSeries.BuildSorted(nextValues, null, null, order, trackChips: true);
            var mask = marks.CreateMask(next);
            Check(Enumerable.Range(0, mask.Length).All(i => mask[i] == (next.RecordAtPoint(i)!.Id is "a" or "retest")),
                "chip marking follows different test values and " + order + " ordering");
            Check(marks.CreateMask(next.Scale(1000)).SequenceEqual(mask), "unit scaling preserves marked identities " + order);
        }
        var filtered = ScatterSeries.Build(new[] { new TestValue(records[4], 999) }, null, null, trackChips: true);
        Check(marks.CreateMask(filtered).Single(), "marks survive a filtered/retest-only series");
        Check(marks.CreateMask(ScatterSeries.Empty).Length == 0 && marks.Count == 1, "temporary empty series preserves chip marks");
        Check(marks.AddRectangle(first, 4, 3, 100, 0) && marks.Count == 3, "reverse rectangle accumulates across wafer and lot");
        marks.Clear();
        Check(marks.Count == 0 && marks.CreateMask(first).Length == 0, "clear releases membership mask requirement");

        var ft = new[]
        {
            new CsvRecord { Id = "f1", IsFtIdentity = true, Lot = "F", RetestUuid = "chip", WaferId = "A", Site = "1" },
            new CsvRecord { Id = "f2", IsFtIdentity = true, Lot = "F", RetestUuid = "chip", WaferId = "B", Site = "2" },
            new CsvRecord { Id = "f3", IsFtIdentity = true, Lot = "G", RetestUuid = "chip" },
            new CsvRecord { Id = "f4", IsFtIdentity = true, Lot = "F" },
            new CsvRecord { Id = "f5", IsFtIdentity = true, Lot = "F" },
            new CsvRecord { Id = "sn1", Lot = "S", WaferId = "W", OriginalSerialNumber = "source", SerialNumber = "1" },
            new CsvRecord { Id = "sn2", Lot = "S", WaferId = "W", OriginalSerialNumber = "source", SerialNumber = "99" },
            new CsvRecord { Id = "unknown1" }, new CsvRecord { Id = "unknown2" }
        };
        var ftSeries = ScatterSeries.Build(ft.Select(r => new TestValue(r, 0)).ToArray(), null, null, trackChips: true);
        marks.AddRectangle(ftSeries, 1, 1, 0, 0);
        Check(marks.CreateMask(ftSeries).SequenceEqual(new[] { true, true, false, false, false, false, false, false, false }),
            "FT UUID crosses site/wafer while remaining isolated by lot");
        marks.AddRectangle(ftSeries, 4, 4, 0, 0);
        Check(marks.CreateMask(ftSeries)[3] && !marks.CreateMask(ftSeries)[4], "missing FT UUID uses unique record identity");
        marks.AddRectangle(ftSeries, 6, 6, 0, 0);
        Check(marks.CreateMask(ftSeries)[6], "CP without XY retains source serial across reassigned Run serials");
        marks.AddRectangle(ftSeries, 8, 8, 0, 0);
        Check(marks.CreateMask(ftSeries)[7] && !marks.CreateMask(ftSeries)[8], "missing CP identity does not merge unrelated records");
        // Exercise the optimized column/view paths, rather than only materialized TestValue arrays.
        var data = Fixture();
        var columnSeries = ScatterSeries.BuildSorted(data.Items[0].Values, 0, 2, ScatterOrder.Site, trackChips: true);
        marks.Clear(); marks.AddRectangle(columnSeries, 1, 1, -100, 100);
        var sparse = ScatterSeries.Build(data.Items[1].Values, null, null, trackChips: true);
        Check(marks.CreateMask(sparse).Single(), "marked CP identity crosses sparse numeric columns");
        var indexed = new IndexedTestValues(data.Items[0].Values, new[] { 2, 3 });
        Check(marks.CreateMask(ScatterSeries.Build(indexed, null, null, trackChips: true)).SequenceEqual(new[] { true, false }),
            "indexed view resolves original row identity");
    }
    static void CompactFailExport()
    {
        var config = Config(); config.PassHBins = new() { "1" }; config.PartialPassHBins = new() { "3" };
        var first = new TestItemData { Id = "a", Name = "Current", DisplayName = "Current", TestNumber = "101", Unit = "A", IsStdfMeasurement = true, LowLimit = 0, HighLimit = .005 };
        var second = new TestItemData { Id = "b", Name = "Voltage", DisplayName = "Voltage", TestNumber = "102", Unit = "V", LowLimit = 10, HighLimit = 15 };
        CsvRecord R(string sn, int x, string hbin, string a, string? b) => new()
        {
            Id = sn, SerialNumber = sn, WaferId = "W", X = x, Y = 0, Site = "1", HBin = hbin, SBin = hbin,
            Measurements = b == null ? new Dictionary<string, string> { ["a"] = a } : new Dictionary<string, string> { ["a"] = a, ["b"] = b }
        };
        var records = new List<CsvRecord>
        {
            R("g1", 0, "1", "0", "10"), R("g2", 1, "1", ".002", "12"), R("g3", 2, "1", ".004", "14"),
            R("f-first", 3, "1", ".001", "11"), R("other-first", 4, "2", ".100", "55"), R("f-last", 3, "2", ".100", "999"),
            R("other-last", 4, "1", ".002", "12"), R("partial", 5, "3", ".007", "17"), R("invalid", 6, "2", "bad", null),
            R("b-first", 7, "2", ".100", "55"), R("b-last", 7, "1", ".002", "12"),
            R("p-first", 8, "1", ".002", "12"), R("p-last", 8, "1", ".002", "12")
        };
        var data = DatasetService.Build(new() { first, second }, records);
        using var lease = new DatasetLease(data.Records); MeasurementDisplay.Configure(data.Items);
        var snapshot = string.Join("|", records.Select(r => r.Id + ":" + string.Join(",", r.Measurements.OrderBy(kv => kv.Key).Select(kv => kv.Key + "=" + kv.Value))));
        string path = P("compact-fail.xlsx");
        var result = FailChipXlsxService.Save(path, data, config, new[] { "W" }, RetestMode.All);
        Check(result.FailChips == 3 && result.RetestGoodChips == 3 && result.Records == 11 && result.ReferenceSerial == "g2", "two-sheet export counts Fail chips, retested Pass chips and reference rows");
        using (var doc = DocumentFormat.OpenXml.Packaging.SpreadsheetDocument.Open(path, false))
        {
            var errors = new DocumentFormat.OpenXml.Validation.OpenXmlValidator().Validate(doc).ToArray();
            foreach (var error in errors) Console.WriteLine(error.Description + " at " + error.Path?.XPath);
            Check(errors.Length == 0, "compact two-sheet XLSX passes Open XML validation");
            foreach (var part in doc.WorkbookPart!.WorksheetParts)
            {
                var pane = part.Worksheet.Descendants<DocumentFormat.OpenXml.Spreadsheet.Pane>().Single();
                Check(pane.HorizontalSplit!.Value == 1 && pane.VerticalSplit!.Value == 1 && pane.TopLeftCell!.Value == "B2", "freeze only test-name row and group column");
                Check(!part.Worksheet.Descendants<DocumentFormat.OpenXml.Spreadsheet.MergeCell>().Any(), "chip titles and explanatory merged rows removed");
                Check(part.Worksheet.Descendants<DocumentFormat.OpenXml.Spreadsheet.Row>().All(r => r.OutlineLevel == null || r.OutlineLevel.Value == 0), "flat chip records avoid obsolete outline groups");
            }
        }
        using (var workbook = new ClosedXML.Excel.XLWorkbook(path))
        {
            Check(workbook.Worksheets.Select(w => w.Name).SequenceEqual(new[] { "Fail 芯片", "复测良品芯片" }), "two sheets have stable descriptive names and order");
            var fail = workbook.Worksheet(1); var good = workbook.Worksheet(2);
            Dictionary<string, ClosedXML.Excel.IXLRow> Rows(ClosedXML.Excel.IXLWorksheet sheet) => sheet.RowsUsed().Where(r => records.Any(s => s.Id == r.Cell(12).GetString())).ToDictionary(r => r.Cell(12).GetString());
            var fs = Rows(fail); var gs = Rows(good);
            foreach (var sheet in workbook.Worksheets)
            {
                Check(sheet.Cell("A1").GetString() == "组别" && sheet.Cell("M1").GetString() == "Current" && sheet.Cell("A2").GetString() == "#TestNumber", "first row is actual column headers without four explanatory rows");
                Check(sheet.Cell("M3").GetString() == "mA" && sheet.Cell("M5").GetDouble() == 5, "unfrozen metadata retains scaled units and limits");
                Check(sheet.RowsUsed().Skip(7).All(r => !r.Cell(12).IsEmpty()) && sheet.RowsUsed().Skip(7).First().RowNumber() == 8, "records start at row 8 without chip titles or spacer rows");
            }
            Check(fs.Keys.Order().SequenceEqual(new[] { "f-first", "f-last", "g2", "invalid", "partial" }.Order()), "Fail sheet preserves original good reference and excludes recovered chips");
            Check(gs.Keys.Order().SequenceEqual(new[] { "other-first", "other-last", "b-first", "b-last", "p-first", "p-last" }.Order()), "good sheet includes all retested final Pass chips, including Pass-to-Pass");
            Check(gs["other-last"].RowNumber() == gs["other-first"].RowNumber() + 1 && fs["f-last"].RowNumber() == fs["f-first"].RowNumber() + 1, "interleaved source history becomes adjacent chip records on both sheets");
            Check(fs["g2"].Cell(1).GetString() == "良品参考" && gs["other-first"].Cell(1).GetString() == gs["other-last"].Cell(1).GetString(), "group labels repeat on each record without separate chip title");
            string Fill(ClosedXML.Excel.IXLRow row, int column) => row.Cell(column).Style.Fill.BackgroundColor.Color.Name;
            Check(Fill(gs["other-last"], 13) == "ffe2f0d9" && Fill(gs["b-last"], 13) == "ffa9d18e" && Fill(gs["p-last"], 13) == "ffe2f0d9", "adjacent chips alternate light and dark green then repeat");
            Check(Fill(gs["other-first"], 13) == "fffce0e8" && Fill(gs["b-first"], 13) == "fff2a0af", "failed measurements use corresponding light and dark red");
            Check(Fill(fs["f-first"], 13) == "ffa9d18e" && Fill(fs["f-last"], 13) == "fff2a0af", "one chip retains its shade across Pass and Fail records");
            Check(Fill(gs["b-first"], 1) == Fill(gs["b-last"], 1) && Fill(gs["other-last"], 1) != Fill(gs["b-first"], 1), "identity columns also distinguish contiguous chip blocks");
            Check(fs["invalid"].Cell(13).GetString() == "bad" && fs["invalid"].Cell(14).IsEmpty() && Fill(fs["invalid"], 13) != Fill(fs["invalid"], 14), "invalid text remains yellow and missing measurements neutral");
            Check(fs["g2"].Cell(13).GetDouble() == 2 && gs["other-last"].Cell(13).GetDouble() == 2, "STDF display scaling remains correct on both sheets");
        }
        foreach (var mode in new[] { RetestMode.First, RetestMode.Last, RetestMode.OnlyRetests })
        {
            string selectedPath = P("compact-" + mode + ".xlsx");
            var selected = FailChipXlsxService.Save(selectedPath, data, config, new[] { "W" }, mode);
            using var book = new ClosedXML.Excel.XLWorkbook(selectedPath);
            var ids = book.Worksheet(2).RowsUsed().Skip(7).Select(r => r.Cell(12).GetString()).ToArray();
            Check(ids.Order().SequenceEqual((mode == RetestMode.First ? new[] { "other-first", "b-first", "p-first" } : new[] { "other-last", "b-last", "p-last" }).Order()), mode + " applies record retention within final-Pass cohort");
            Check(selected.RetestGoodChips == 3 && selected.Records == (mode == RetestMode.OnlyRetests ? 5 : 7), mode + " returns actual written record counts including reference");
        }
        var noGood = new CsvParseResult { Items = data.Items, Records = records.Where(r => r.X is 3 or 5).ToList() };
        string noGoodPath = P("compact-no-good.xlsx");
        var failOnly = FailChipXlsxService.Save(noGoodPath, noGood, config, new[] { "W" }, RetestMode.All);
        using (var book = new ClosedXML.Excel.XLWorkbook(noGoodPath))
            Check(book.Worksheets.Count == 2 && book.Worksheet(2).LastRowUsed()!.RowNumber() == 7 && failOnly.ReferenceSerial == null && failOnly.RetestGoodChips == 0, "empty good cohort still writes its header and metadata sheet");
        string allGoodPath = P("compact-all-retested-good.xlsx");
        var allGood = FailChipXlsxService.Save(allGoodPath, new CsvParseResult { Items = data.Items, Records = records.Where(r => r.X == 4).ToList() }, config, new[] { "W" }, RetestMode.All);
        using (var book = new ClosedXML.Excel.XLWorkbook(allGoodPath))
            Check(allGood.FailChips == 0 && allGood.RetestGoodChips == 1 && allGood.Records == 2 && book.Worksheet(1).LastRowUsed()!.RowNumber() == 7, "retested-good-only population exports with an empty Fail sheet");
        string unchanged = P("compact-keep.xlsx"); File.WriteAllText(unchanged, "keep");
        Reject<InvalidDataException>(() => FailChipXlsxService.Save(unchanged, new CsvParseResult { Items = data.Items, Records = records.Where(r => r.X < 3).ToList() }, config, new[] { "W" }, RetestMode.All), "no Fail or retested-good population is rejected");
        Check(File.ReadAllText(unchanged) == "keep", "empty export preserves existing destination");
        Reject<InvalidDataException>(() => FailChipXlsxService.Save(P("invalid-wafer.xlsx"), data, config, new[] { "missing" }, RetestMode.All), "export validates selected wafers");
        Reject<InvalidDataException>(() => FailChipXlsxService.Save(P("invalid-sigma.xlsx"), data, config, new[] { "W" }, RetestMode.All, 0), "export validates optional outlier sigma");
        var ftRecords = records.Where(r => r.X is 3 or 4).Select((r, i) => new CsvRecord
        {
            Id = r.Id, SerialNumber = r.SerialNumber, WaferId = i < 2 ? "W" : "W2", Lot = "L", IsFtIdentity = true,
            RetestUuid = r.X == 3 ? "fail-chip" : "good-chip", X = i * 20, Y = 0, HBin = r.HBin, Measurements = r.Measurements
        }).ToList();
        string ftPath = P("compact-ft.xlsx");
        var ft = FailChipXlsxService.Save(ftPath, new CsvParseResult { Items = data.Items, Records = ftRecords }, config, new[] { "W", "W2" }, RetestMode.All);
        using (var book = new ClosedXML.Excel.XLWorkbook(ftPath))
            Check(ft.FailChips == 1 && ft.RetestGoodChips == 1 && book.Worksheet(2).RowsUsed().Skip(7).Count() == 2, "FT UUID follows one chip across wafer and XY changes into correct sheet");
        string clippedPath = P("compact-outliers.xlsx");
        FailChipXlsxService.Save(clippedPath, data, config, new[] { "W" }, RetestMode.All, 1);
        using (var book = new ClosedXML.Excel.XLWorkbook(clippedPath))
        {
            var clipped = book.Worksheet(1).RowsUsed().Single(r => r.Cell(12).GetString() == "f-last");
            Check(clipped.Cell(13).IsEmpty() && clipped.Cell(13).Style.Fill.BackgroundColor.Color.Name == "ffd9dee3", "outlier clipping uses the current chip shade's neutral fill");
        }
        string after = string.Join("|", records.Select(r => r.Id + ":" + string.Join(",", r.Measurements.OrderBy(kv => kv.Key).Select(kv => kv.Key + "=" + kv.Value))));
        Check(snapshot == after && first.Values[1].Value == .002, "all export modes leave source measurements intact");
        using var canceled = new CancellationTokenSource(); canceled.Cancel();
        using (new AnalysisOperation(canceled.Token))
            Reject<OperationCanceledException>(() => FailChipXlsxService.Save(P("compact-cancelled.xlsx"), data, config, new[] { "W" }, RetestMode.All), "two-sheet export respects cancellation");
        Check(!File.Exists(P("compact-cancelled.xlsx")) && !Directory.GetFiles(folder, "compact-*.tmp").Any(), "cancelled/failed exports leave no partial output or temp files");
    }
